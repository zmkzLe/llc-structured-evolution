===== STRUCTURE_DESCRIPTION =====
name: mj_refined_confidence_eviction
doc: >
  Refines the confidence training logic to more accurately track the RDP's prediction
  of deadness, aiming to reduce `wrong_bypass`. Confidence in a line being 'dead'
  (long-reuse) is incremented only if the RDP also predicted long reuse, and decremented
  if RDP predicted short reuse but the line was found dead in the sampler. Additionally,
  a new rule in `procedures.access` leverages this refined confidence to aggressively
  mark lines for eviction (`etr=11`) if their signature is confidently predicted as dead,
  which should help reduce `kept_dead` mistakes. The sampler size remains at 5 ways.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when: # Bypass when RDP predicts long reuse AND confidence is high.
    - confident_rdp_greater_than: {conf_table: rdp_conf, rdp_table: rdp, index: pc_sig, threshold: 73}
    - confident_table_exceeds_victim: {conf_table: rdp_conf, rdp_table: rdp, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_conf # Tracks confidence in RDP's scan prediction.
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5 # Restore original Mockingjay sampler size.
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # A reuse was detected -> the line was not dead.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence in "dead" prediction.
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep: # A line aged out without reuse -> likely dead.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}, rdp_compare: {rdp_table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: increment} # Increment confidence in "dead" prediction if RDP also predicted dead.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}, rdp_compare: {rdp_table: rdp, index: signature, cmp: lte, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence if RDP predicted alive but it died.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace: # A line was evicted without reuse -> likely dead.
      - when: {rdp_compare: {rdp_table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: increment} # Increment confidence in "dead" prediction if RDP also predicted dead.
      - when: {rdp_compare: {rdp_table: rdp, index: signature, cmp: lte, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence if RDP predicted alive but it was replaced.
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    # New rule: if confidence in "dead" prediction is high, evict quickly.
    - when: {table: {table: rdp_conf, index: pc_sig, cmp: is_max, threshold: MAX}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_rdp_greater_than
  place: condition
  params: {conf_table: table, rdp_table: table, index: signature, threshold: {int: [0, 127]}}
  means: >
    True if the entry for the index in conf_table is present and at its maximum value, AND
    the entry for the same index in rdp_table is present and has a value greater than the threshold.
    False otherwise.
- name: confident_table_exceeds_victim
  place: condition
  params: {conf_table: table, rdp_table: table, index: signature, div: {int: [1, 1024]}}
  means: >
    True if the entry for the index in conf_table is present and at its maximum value, AND
    the value from rdp_table[index]/div (integer division) is greater than the absolute
    value of the victim key of the way the victim search chose. An absent entry in rdp_table
    is treated as a value of 0. False otherwise.
- name: rdp_compare
  place: condition
  params: {rdp_table: table, index: signature, cmp: {symbol: [Comparison]}, threshold: {int: [0, 127]}}
  means: >
    True if the entry for the index in rdp_table is present and its value satisfies the comparison
    with the threshold. An absent entry acts as 0 for comparison.
