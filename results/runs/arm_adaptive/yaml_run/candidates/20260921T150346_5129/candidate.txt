name: mj_hybrid_confidence_bypass
doc: >
  This design uses a hybrid bypass strategy to balance wrong_bypass and should_bypass
  errors. The static bypass condition (RDP > 73) is gated by maximum confidence
  (3/3) to prevent incorrect dead-on-arrival predictions. The dynamic bypass condition,
  which compares the incoming line's predicted reuse against the victim's ETR, is
  not gated by confidence. This allows it to be more responsive and should reduce
  the high number of should_bypass errors seen in the previous, more restrictive design,
  while retaining the benefits of confidence for the static prediction.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {confident_bypass_rdp: {rdp_table: rdp, conf_table: rdp_confidence, index: pc_sig, rdp_thresh: 73, conf_thresh: 3}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 4}
      - when: {type_not_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 4}
      - train: {table: rdp_confidence, index: signature, op: increment}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_confidence, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_confidence, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 4}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}
    - when: {table: {table: rdp_confidence, index: pc_sig, cmp: lt, threshold: 2}}
      set: {etr: {add: 2}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
