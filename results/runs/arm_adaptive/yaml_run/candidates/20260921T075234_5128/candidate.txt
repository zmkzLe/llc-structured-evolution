===== STRUCTURE_DESCRIPTION =====
name: mj_conf_bypass
doc: >
  Mockingjay with a confidence mechanism for its bypass predictor, to reduce
  wrong bypasses. A 2-bit confidence counter is associated with each entry in
  the reuse distance predictor (RDP) table. Confidence is increased upon a
  measured reuse in the sampler, and decreased when a line is evicted from
  the sampler without reuse (which Mockingjay treats as a scan). A bypass
  decision now requires both a long predicted reuse distance and maximum confidence.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # A bypass requires a long predicted reuse and high confidence
    - confident_long_reuse_pred: {rdp: rdp, conf: rdp_conf, index: pc_sig, rdp_thresh: 73}
    - confident_table_exceeds_victim: {rdp: rdp, conf: rdp_conf, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train_rdp_toward_conf_inc: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train_rdp_toward_conf_inc: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train_rdp_inc_conf_dec: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
        free: true
    on_replace:
      - train_rdp_inc_conf_dec: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the CRC2 update body, shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_long_reuse_pred
  place: condition
  params: {rdp: table, conf: table, index: signature, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the rdp table entry at the index is present and its value is greater than
    rdp_thresh, AND the conf table entry at the same index is present and its value
    is its maximum value (2^width - 1). False otherwise.

- name: confident_table_exceeds_victim
  place: condition
  params: {rdp: table, conf: table, index: signature, div: {int: [1, 128]}}
  means: >
    True when the rdp table entry at the index is present, its value divided by div
    (integer division) is greater than the absolute value of the chosen victim's key,
    AND the conf table entry at the index is present and its value is its maximum value
    (2^width - 1). False otherwise. Only for use in bypass_when.

- name: train_rdp_toward_conf_inc
  place: step
  params: {rdp_table: table, conf_table: table, index: signature, sample: {symbol: [age]}, scale: {int: [1, 10]}, min_diff: {int: [0, 127]}}
  means: >
    Trains the rdp_table toward the sample value (age), scaled by scale, with a minimum
    difference of min_diff. Simultaneously, increments the conf_table entry at the same index.
    The rdp_table training is identical to the 'toward' TrainOp. The conf_table training is
    identical to the 'increment' TrainOp.

- name: train_rdp_inc_conf_dec
  place: step
  params: {rdp_table: table, conf_table: table, index: signature}
  means: >
    Increments the rdp_table entry at the given index. Simultaneously, decrements
    the conf_table entry at the same index. The training for each table is identical
    to the 'increment' and 'decrement' TrainOp respectively.
