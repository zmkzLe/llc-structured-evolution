name: mj_conf_bypass_v2
doc: >
  Corrected the confidence training logic: confidence is now incremented when a line
  is observed to be dead (evicted from sampler without reuse) and decremented when
  reused. This aligns `rdp_conf` with 'confidence in long reuse distance' predictions,
  aiming to fix `wrong_bypass` and `kept_dead`. Renamed custom condition words
  to avoid clashes with previously accepted words. The bypass and eviction logic
  remain the same, utilizing this corrected confidence.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # A bypass requires a long predicted reuse and high confidence
    - my_confident_rdp_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, rdp_thresh: 80} # Stricter threshold for bypass
    - my_confident_rdp_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 10} # Stricter divisor for bypass

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train_rdp_toward_conf_dec: {rdp_table: rdp, conf_table: rdp_conf, index: signature, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train_rdp_toward_conf_dec: {rdp_table: rdp, conf_table: rdp_conf, index: signature, scale: 1, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train_rdp_inc_conf_inc: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
        free: true
    on_replace:
      - train_rdp_inc_conf_inc: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the CRC2 update body, shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {my_confident_rdp_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, rdp_thresh: 73}} # Set etr=11 for confidently dead blocks
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: my_confident_rdp_gt_threshold
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the rdp_table entry at the index is present and its value is greater than
    rdp_thresh, AND the conf_table entry at the same index is present and its value
    is its maximum value (2^width - 1). False otherwise.

- name: my_confident_rdp_exceeds_victim
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, div: {int: [1, 128]}}
  means: >
    True when the rdp_table entry at the index is present, its value divided by div
    (integer division) is greater than the absolute value of the chosen victim's key,
    AND the conf_table entry at the index is present and its value is its maximum value
    (2^width - 1). False otherwise. Only for use in bypass_when.

- name: train_rdp_toward_conf_dec
  place: step
  params: {rdp_table: table, conf_table: table, index: signature, scale: {int: [1, 10]}, min_diff: {int: [0, 127]}}
  means: >
    Trains the rdp_table toward the value 'age', scaled by scale, with a minimum
    difference of min_diff. Simultaneously, decrements the conf_table entry at the same index.
    The rdp_table training is identical to the 'toward' TrainOp. The conf_table training is
    identical to the 'decrement' TrainOp.

- name: train_rdp_inc_conf_inc
  place: step
  params: {rdp_table: table, conf_table: table, index: signature}
  means: >
    Increments the rdp_table entry at the given index. Simultaneously, increments
    the conf_table entry at the same index. The training for each table is identical
    to the 'increment' TrainOp.
