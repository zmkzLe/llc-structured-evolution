===== STRUCTURE_DESCRIPTION =====
name: mockingjay_fastlearn_conf_bypass_fix
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. A 2-bit confidence counter is added per
  RDP entry, incremented on sampled reuses and decremented on misses. Bypass decisions
  now require a minimum confidence level (3 out of 3), aiming to reduce wrong bypasses.
  Writebacks are inserted at -INF_ETR and train nothing.

  This version fixes previous syntax errors related to multiple 'train' actions in a
  single rule by introducing new helper words. It also incorporates faster learning
  by reducing the RDP min_diff from 16 to 4, and sets a more conservative initial
  ETR of 4 for absent RDP entries (was 0), aiming to reduce both wrong_bypass and
  kept_dead mistakes.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {confident_bypass_rdp: {rdp_table: rdp, conf_table: rdp_confidence, index: pc_sig, rdp_thresh: 73, conf_thresh: 3}}
    - {confident_bypass_victim: {rdp_table: rdp, conf_table: rdp_confidence, index: pc_sig, div: 8, conf_thresh: 3}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2                  # 0-3 confidence level
    init: 0

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train_rdp_toward_conf_inc: {rdp_table: rdp, conf_table: rdp_confidence, index: signature, sample: age, scale: 2, min_diff: 4} # min_diff changed from 16 to 4
        free: true
        stop: true
      - train_rdp_toward_conf_inc: {rdp_table: rdp, conf_table: rdp_confidence, index: signature, sample: age, scale: 1, min_diff: 4} # min_diff changed from 16 to 4, explicit scale 1
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train_rdp_inc_conf_dec: {rdp_table: rdp, conf_table: rdp_confidence, index: signature}
        free: true
    on_replace:
      - train_rdp_inc_conf_dec: {rdp_table: rdp, conf_table: rdp_confidence, index: signature}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the CRC2 update body, shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 4}} # Changed from 0 to 4
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: train_rdp_toward_conf_inc
  place: step
  params:
    rdp_table: table
    conf_table: table
    index: signature
    sample: sampler
    scale: {int: [1, 10]}
    min_diff: {int: [0, 127]}
  means: >
    Trains the rdp_table toward the value 'sample', scaled by scale, with a minimum difference of
    min_diff. Simultaneously, increments the conf_table entry at the same index. The rdp_table training
    is identical to the 'toward' TrainOp. The conf_table training is identical to the 'increment'
    TrainOp.

- name: train_rdp_inc_conf_dec
  place: step
  params: {rdp_table: table, conf_table: table, index: signature}
  means: >
    Increments the rdp_table entry at the given index. Simultaneously, decrements the conf_table
    entry at the same index. The training for each table is identical to the 'increment' and 'decrement'
    TrainOp respectively.
