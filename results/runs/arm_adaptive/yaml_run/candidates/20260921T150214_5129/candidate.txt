name: mj_relative_bypass
doc: >
  To reduce wrong_bypass mistakes, this design introduces a more intelligent,
  relative bypass condition. Instead of comparing the incoming line's predicted
  reuse distance to a fixed threshold, it compares it to the predicted reuse
  distance of the victim line. A new condition word,
  'incoming_rdp_worse_than_victim_rdp', is defined for this purpose. A line is
  bypassed only if it is confidently predicted to be less useful (longer reuse
  distance) than the line it would evict. This is expected to make bypass
  decisions more context-aware and reduce incorrect bypasses.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - incoming_rdp_worse_than_victim_rdp: {rdp_table: rdp, conf_table: rdp_conf, incoming_index: pc_sig}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_conf # Tracks confidence in RDP's scan prediction.
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # A reuse was detected.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence in scan.
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep: # A line aged out without reuse (scan-like behavior).
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: increment} # Increment confidence in scan.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace: # A line was evicted without reuse (scan).
      - train: {table: rdp_conf, index: signature, op: increment} # Increment confidence in scan.
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp_conf, index: pc_sig, cmp: is_max, threshold: MAX}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: incoming_rdp_worse_than_victim_rdp
  place: condition
  params: {rdp_table: table, conf_table: table, incoming_index: signature}
  means: >
    This condition is only for use in `bypass_when`. It first identifies the victim way
    chosen by the victim search. It then reads the Instruction Pointer (IP) of the line
    in the victim way. It computes a signature for this victim IP (`victim_index`) using
    the same signature definition (source, folding, hash, etc.) and access context
    (hit, prefetch) as the `incoming_index`. It then reads the `rdp_table` for both
    the `incoming_index` and the `victim_index`. An absent entry is treated as a value of 0.
    The condition is true if `rdp_table[incoming_index] > rdp_table[victim_index]` AND the
    entry in `conf_table` for the `incoming_index` is present and at its maximum value. False otherwise.
