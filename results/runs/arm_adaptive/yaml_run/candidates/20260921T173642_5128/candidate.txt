name: mj_fix_conf_and_bypass_cmp
doc: >
  Fixes the inverted confidence training logic of the parent and introduces a more
  accurate bypass condition to address 'wrong_bypass' errors. Confidence in a
  line being 'dead' is now correctly learned (incremented on no-reuse, decremented
  on reuse). The bypass decision is now based on a direct comparison between the
  incoming line's predicted reuse distance (RDP) and an estimate of the victim's
  RDP, gated by high confidence. This prevents bypassing lines that are more
  valuable than the victim.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when: # Bypass if confident incoming is worse than victim
    - confident_incoming_rdp_worse_than_victim_etr: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, scale: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_conf # Confidence in 'dead' prediction. High value means confident it's dead.
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # Reuse detected: line is not dead. Decrement confidence.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_rdp_toward_conf_dec_v2: {rdp_table: rdp, conf_table: rdp_conf, index: signature, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train_rdp_toward_conf_dec_v2: {rdp_table: rdp, conf_table: rdp_conf, index: signature, scale: 1, min_diff: 16}
        free: true
    sweep: # Aged out without reuse: likely dead. Increment confidence.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train_rdp_inc_conf_inc_v2: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
        free: true
    on_replace: # Evicted without reuse: likely dead. Increment confidence.
      - train_rdp_inc_conf_inc_v2: {rdp_table: rdp, conf_table: rdp_conf, index: signature}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {my_confident_long_reuse_pred: {rdp: rdp, conf: rdp_conf, index: pc_sig, rdp_thresh: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: my_confident_long_reuse_pred
  place: condition
  params: {rdp: table, conf: table, index: signature, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the rdp table entry at the index is present and its value is greater than
    rdp_thresh, AND the conf table entry at the same index is present and its value
    is its maximum value (2^width - 1). False otherwise.
- name: confident_incoming_rdp_worse_than_victim_etr
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, scale: {int: [1, 16]}}
  means: >
    Reads the victim key of the chosen victim way, which is 'etr'. Reads the rdp_table entry
    at the given index. Let the victim's key be V and the table value be T. True if the
    entry in conf_table at the index is present and at its maximum value, AND T is greater
    than the absolute value of V multiplied by scale. An absent rdp_table entry is treated as
    a value of 0. This word can only be used in bypass_when.
- name: train_rdp_toward_conf_dec_v2
  place: step
  params: {rdp_table: table, conf_table: table, index: signature, scale: {int: [1, 10]}, min_diff: {int: [0, 127]}}
  means: >
    Trains the rdp_table toward the value 'age', scaled by scale, with a minimum
    difference of min_diff. Simultaneously, decrements the conf_table entry at the same index.
    The rdp_table training is identical to the 'toward' TrainOp. The conf_table training is
    identical to the 'decrement' TrainOp.
- name: train_rdp_inc_conf_inc_v2
  place: step
  params: {rdp_table: table, conf_table: table, index: signature}
  means: >
    Increments the rdp_table entry at the given index. Simultaneously, increments
    the conf_table entry at the same index. The training for each table is identical
    to the 'increment' TrainOp.
