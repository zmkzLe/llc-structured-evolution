===== STRUCTURE_DESCRIPTION =====
name: mj_conservative_rdp_thresholds
doc: >
  To reduce wrong_bypass and evicted_sooner mistakes, this design makes the
  absolute RDP thresholds more conservative. The threshold for bypassing based
  on a high RDP value, and for explicitly setting ETR to its maximum (11),
  is increased from 73 to 80. This means only lines confidently predicted to have
  an RDP in the highest bucket (81-95) are subject to these aggressive actions,
  reducing premature bypasses and evictions of lines that are merely "bad" but
  not "worst". The bypass condition that compares incoming RDP against the
  victim's ETR remains unchanged as it is already relative.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when: # With corrected confidence, this now bypasses confidently-predicted dead lines
    - confident_rdp_greater_than: {conf_table: rdp_conf, rdp_table: rdp, index: pc_sig, threshold: 80} # Changed from 73
    - confident_table_exceeds_victim: {conf_table: rdp_conf, rdp_table: rdp, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_conf # Confidence in 'dead' prediction. High value means confident it's dead.
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5 # Increased from 4 back to 5 for better accuracy
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # Reuse detected: line is not dead. Decrement confidence.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep: # Aged out without reuse: likely dead. Increment confidence.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace: # Evicted without reuse: likely dead. Increment confidence.
      - train: {table: rdp_conf, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp_conf, index: pc_sig, cmp: is_max, threshold: MAX}}
      set: {etr: {const: 11}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 80}} # Changed from 73
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_rdp_greater_than
  place: condition
  params: {conf_table: table, rdp_table: table, index: signature, threshold: {int: [0, 127]}}
  means: >
    True if the entry for the index in conf_table is present and at its maximum value, AND
    the entry for the same index in rdp_table is present and has a value greater than the threshold.
    False otherwise.
- name: confident_table_exceeds_victim
  place: condition
  params: {conf_table: table, rdp_table: table, index: signature, div: {int: [1, 1024]}}
  means: >
    True if the entry for the index in conf_table is present and at its maximum value, AND
    the value from rdp_table[index]/div (integer division) is greater than the absolute
    value of the victim key of the way the victim search chose. An absent entry in rdp_table
    is treated as a value of 0. False otherwise.
