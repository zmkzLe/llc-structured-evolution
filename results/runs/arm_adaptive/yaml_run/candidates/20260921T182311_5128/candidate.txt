===== STRUCTURE_DESCRIPTION =====
name: mj_dynamic_bypass_revert
doc: >
  Reverts the flawed confidence-based bypass and eviction logic. This design removes the rdp_conf
  table and its associated logic, which the guidance identified as detrimental. It adopts a more direct
  and dynamic bypass condition, `incoming_rdp_worse_than_victim`, which was previously successful.
  This condition compares the predicted reuse distance of the incoming line against that of the
  victim, aiming for a more accurate decision and a reduction in `wrong_bypass` errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass if the incoming line is predicted to be reused much later than the victim.
    - incoming_rdp_worse_than_victim: {rdp_table: rdp, index: pc_sig, multiplier: 8}
    # Also bypass lines with extremely long predicted reuse, regardless of the victim.
    - table: {table: rdp, index: pc_sig, cmp: gt, threshold: 85}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # A reuse was detected.
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep: # A line aged out without reuse (scan-like behavior).
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace: # A line was evicted without reuse (scan).
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: incoming_rdp_worse_than_victim
  place: condition
  params:
    index: signature
    multiplier:
      int: [1, 16]
    rdp_table: table
  means: >
    Reads the rdp_table entry at the given index; if absent, the value is 0. This is the incoming
    line's predicted RDP value. Reads the victim key of the chosen victim way. This is the victim_etr.
    Calculates a threshold as the absolute value of victim_etr multiplied by the multiplier, to approximate
    the victim's original RDP value. True if the incoming line's predicted RDP value is greater than this
    calculated threshold. False otherwise. This word can only be used in a 'bypass_when' clause.
