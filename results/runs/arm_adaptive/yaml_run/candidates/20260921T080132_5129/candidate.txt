name: mj_fix_conf_train
doc: >
  Fixes the confidence training logic of previous attempts. This version of
  Mockingjay with a confidence-based bypass now correctly trains its confidence
  predictor (rdp_conf). Confidence is INCREASED on evidence of a scan (an eviction
  from the sampler without reuse) and DECREASED on evidence of reuse (a match).
  A high confidence value thus correctly indicates a scanning access pattern. The
  bypass decision requires both a long predicted reuse distance from the RDP and
  maximum confidence, making the bypass logic sound.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - confident_long_reuse_pred: {rdp: rdp, conf: rdp_conf, index: pc_sig, rdp_thresh: 73}
    - confident_table_exceeds_victim: {rdp: rdp, conf: rdp_conf, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp_conf, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_long_reuse_pred
  place: condition
  params: {rdp: table, conf: table, index: signature, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the rdp table entry at the index is present and its value is greater than
    rdp_thresh, AND the conf table entry at the same index is present and its value
    is its maximum value (2^width - 1). False otherwise.

- name: confident_table_exceeds_victim
  place: condition
  params: {rdp: table, conf: table, index: signature, div: {int: [1, 128]}}
  means: >
    True when the rdp table entry at the index is present, its value divided by div
    (integer division) is greater than the absolute value of the chosen victim's key,
    AND the conf table entry at the index is present and its value is its maximum value
    (2^width - 1). False otherwise. Only for use in bypass_when.
