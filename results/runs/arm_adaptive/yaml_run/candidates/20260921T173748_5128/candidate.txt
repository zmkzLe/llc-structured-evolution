name: mj_confident_bypass_and_penalty_v3
doc: >
  Refines bypass logic and learning parameters to address high 'wrong_bypass' and
  'kept_dead' errors. The `table_exceeds_victim` bypass condition is now confidence-aware,
  requiring maximum confidence for the incoming line's RDP prediction to bypass.
  RDP training `min_diff` is reduced from 16 to 4 for faster adaptation, and the
  initial ETR for unseen signatures is set to 4 (instead of 0) for a more conservative
  initial prediction. These changes aim to make bypass decisions more accurate and
  improve learning speed, while the ETR penalty for low-confidence lines is retained.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Replaced table_exceeds_victim with a confidence-aware version, requiring max confidence.
    - confident_table_exceeds_victim: {conf_table: rdp_confidence, index: pc_sig, rdp_table: rdp, div: 8}
    # Retain the confidence-aware RDP threshold bypass.
    - confident_bypass_rdp: {conf_table: rdp_confidence, conf_thresh: 2, index: pc_sig, rdp_table: rdp, rdp_thresh: 73}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: rdp_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_confidence, index: signature, op: decrement}
        free: true
        stop: true
      - when: {type_in: [PREFETCH]}
        # Changed min_diff from 16 to 4 for faster adaptation
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 4}
      - when: {type_not_in: [PREFETCH]}
        # Changed min_diff from 16 to 4 for faster adaptation
        train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 4}
      - train: {table: rdp_confidence, index: signature, op: increment}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_confidence, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_confidence, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      # Changed initial ETR from 0 to 4 for a more conservative prediction
      set: {etr: {const: 4}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}
    - when: {table: {table: rdp_confidence, index: pc_sig, cmp: lt, threshold: 2}}
      set: {etr: {add: 2}}

on_hit:

    conf_table: table

      int: [0, 127]
  means: 'True when the entry in rdp_table at the given index is present and its value is greater than rdp_thresh, AND the entry in conf_table at the given index has a value greater than or equal to conf_thresh. False if the rdp_table entry is absent. An entry in conf_table always exists and is initialized to zero.'
- name: confident_table_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 1024]
    index: signature
    rdp_table: table
  means: True if the entry for the index in conf_table is present and at its maximum value, AND the value from rdp_table[index]/div (integer division) is greater than the absolute value of the victim key of the way the victim search chose. An absent entry in rdp_table is treated as a value of 0. False otherwise.
