===== STRUCTURE_DESCRIPTION =====
name: mj_fast_adapt
doc: >
  Addresses massive wrong_bypass and should_bypass mistakes by replacing the slow
  single-step 'toward' training with a fast-adapting mechanism. When the predicted
  reuse distance is off by a large margin (>32), the predictor jumps by 8 to quickly
  recover, preventing signatures from getting stuck at 95 (causing wrong bypasses)
  or at 0 (causing cache pollution). Additionally, lines replaced without reuse
  increment the predictor by 2 instead of 1, learning dead lines twice as fast.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - words: {train_fast_adapt: {table: rdp, index: signature, age: age, margin: 32, jump: 8, min_diff: 16, max_val: 95}}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        words: {train_add: {table: rdp, index: signature, amount: 2, max_val: 95}}
        free: true
    on_replace:
      - words: {train_add: {table: rdp, index: signature, amount: 2, max_val: 95}}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: train_fast_adapt
  place: step
  params:
    table: table
    index: signature
    age: state
    margin: {int: [1, 127]}
    jump: {int: [1, 127]}
    min_diff: {int: [1, 127]}
    max_val: {int: [1, 127]}
  means: >
    Reads the table entry at the index.
    If the entry is absent, it is set to the age, clamped at max_val.
    Else, if the entry is strictly greater than the age plus the margin, the entry is decreased by the jump value, clamped at 0.
    Else, if the age is strictly greater than the entry plus the margin, the entry is increased by the jump value, clamped at max_val.
    Else, if the absolute difference between the entry and the age is at least min_diff, the entry is incremented by 1 if age > entry, or decremented by 1 if age < entry, clamped between 0 and max_val.
    The final value is written to the table entry.

- name: train_add
  place: step
  params:
    table: table
    index: signature
    amount: {int: [1, 127]}
    max_val: {int: [1, 127]}
  means: >
    Reads the table entry at the index.
    If the entry is absent, it is set to the amount, clamped at max_val.
    Else, the entry is increased by the amount, clamped at max_val.
    The final value is written to the table entry.
