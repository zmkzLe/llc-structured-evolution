===== STRUCTURE_DESCRIPTION =====
name: mockingjay_safe
doc: >
  Mockingjay with a usefulness confidence table to prevent wrong bypasses.
  The useful_conf table (2-bit, init MID) increments on sampler hits and
  decrements on sampler evictions or age-outs. Bypasses only occur if
  useful_conf is 0, ensuring a signature is consistently dead before bypassing.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {bypass_if_dead: {rdp: rdp, conf: useful_conf, index: pc_sig, threshold: 73}}
    - {bypass_exceeds_if_dead: {rdp: rdp, conf: useful_conf, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: useful_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: MID

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
      - when: {type_in: [PREFETCH]}
        train: {table: useful_conf, index: signature, op: increment}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
      - train: {table: useful_conf, index: signature, op: increment}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: useful_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: useful_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: bypass_if_dead
  place: condition
  params:
    rdp: table
    conf: table
    index: signature
    threshold: {int: [0, 127]}
  means: >
    True when the entry of the rdp table at the index is present and its value is strictly greater than the threshold, and the entry of the conf table at the index is equal to 0. False otherwise.

- name: bypass_exceeds_if_dead
  place: condition
  params:
    rdp: table
    conf: table
    index: signature
    div: {int: [1, 16]}
  means: >
    True when the entry of the rdp table at the index is present, its value divided by div (integer division) is strictly greater than the absolute value of the chosen victim way's key, and the entry of the conf table at the index is equal to 0. False on an absent entry. Only inside victim.bypass_when.
