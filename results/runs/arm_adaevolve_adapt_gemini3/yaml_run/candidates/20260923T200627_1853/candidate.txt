===== STRUCTURE_DESCRIPTION =====
name: mockingjay_conf
doc: >
  Mockingjay with confidence-based bypass. A 2-bit confidence table tracks whether
  a signature is consistently dead (incremented on sampler replacement/sweep,
  decremented on match). Fills are only bypassed if the confidence is > 1,
  preventing wrong_bypass mistakes for signatures with mixed reuse behavior.
  Victim extremum is reverted to max_magnitude to avoid performance loss.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {confident_bypass: {conf: conf, index: pc_sig, table: rdp, threshold: 73}}
    - {confident_exceeds_victim: {conf: conf, div: 8, index: pc_sig, table: rdp}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    max: 3
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        train: {table: conf, index: signature, op: decrement}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        train: {table: conf, index: signature, op: decrement}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        train: {table: conf, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: conf, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: confident_bypass
  place: condition
  params:
    conf: table
    index: signature
    table: table
    threshold: {int: [0, 127]}
  means: >
    True when the entry of the table at the index is present and its value is greater than the threshold,
    and the entry of the conf table at the index is greater than 1. False otherwise.

- name: confident_exceeds_victim
  place: condition
  params:
    conf: table
    div: {int: [1, 16]}
    index: signature
    table: table
  means: >
    True when the entry of the table at the index is present, its value divided by div (integer division)
    is greater than the absolute value of the victim key of the way the victim search chose, and the entry
    of the conf table at the index is greater than 1. False on an absent entry. Only inside victim.bypass_when.
