name: mj_unified_rank_deconf_v2
doc: >
  Uses a unified, confidence-penalized rank for both insertion and bypass decisions.
  This aims to resolve the conflict between wrong_bypasses and kept_dead lines
  by applying a consistent eviction priority logic everywhere. The rank is
  calculated from the predicted reuse distance (RDP), discounted by a confidence
  score. This follows the successful `deconfident` pattern, where low confidence
  penalizes the RDP value, making both insertion and bypass more conservative.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - deconfident_rdp_gt:
        rdp_table: rdp
        conf_table: rdp_confidence
        index: pc_sig
        scale: 4
        threshold: 73
    - deconfident_exceeds_victim:
        rdp_table: rdp
        conf_table: rdp_confidence
        index: pc_sig
        div: 8
        scale: 4

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: rdp_confidence, index: signature, op: increment}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_confidence, index: signature, op: decrement}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp_confidence, index: signature, op: decrement}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}

            conf_table: rdp_confidence
            index: pc_sig
            div: 8
            scale: 4

on_hit:

- name: penalized_confidence_etr_v2
  place: value
  params: {rdp_table: table, conf_table: table, index: signature, div: {int: [1, 128]}, scale: {int: [0, 16]}, max_etr: {int: [0, 15]}}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. If rdp_val is greater than 73, returns max_etr.
    'conf_val' is read from its table, with an absent entry read as its init value (0).
    The maximum value for confidence is assumed to be 3. Calculates an adjusted RDP value:
    adjusted_rdp = rdp_val - ((3 - conf_val) * scale).
    If the result of the subtraction is negative, it is treated as 0.
    Calculates a predicted_etr value as adjusted_rdp divided by div, using integer division.
    Returns min(max_etr, predicted_etr).
