name: mj_robust_bypass
doc: >
  Introduces a more robust bypass condition, `bypass_if_worse_than_all`, which compares
  the incoming line's rank against all resident lines. This replaces the previous,
  more aggressive bypass heuristics to specifically target and reduce `wrong_bypass` errors,
  as suggested by the run guidance.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass only if the incoming line is predicted to be worse than all resident lines.
    - bypass_if_worse_than_all: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Calculate ETR using the unified rank function for all lines
    - set:
        etr:
          unified_etr_rank:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: unified_etr_rank
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If 'conf_val' is 0, the returned value is max_etr.
    Otherwise, calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: bypass_if_worse_than_all
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Calculates the rank of the incoming line (incoming_rank) using the same logic as the
    'unified_etr_rank' value word. This is: Read 'rdp_val' from rdp_table[index] and
    'conf_val' from conf_table[index], treating absent as 0. If 'conf_val' is 0,
    incoming_rank is max_etr. Otherwise, base_etr = rdp_val / div, penalty = (3 - conf_val) * scale,
    and incoming_rank = min(max_etr, base_etr + penalty). The condition then iterates through
    all resident ways in the set. It is true if and only if for every resident line,
    the absolute value of incoming_rank is greater than the absolute value of the resident line's
    'etr' field. If the set is empty or any resident line has a rank that is worse or equal,
    the condition is false.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
