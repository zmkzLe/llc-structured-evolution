name: mj_unified_deconfident_etr_v2
doc: >
  Unifies the Estimated Time to Reuse (ETR) calculation for both resident and
  incoming lines by consistently applying a 'deconfident' RDP value. This
  design replaces the previous ETR calculation with a new value word,
  'deconfident_etr_val', which computes ETR by penalizing the RDP value
  based on confidence (lower confidence reduces the effective RDP, making the
  line less evictable). This strategy, derived from top-performing designs,
  aims to reduce 'wrong_bypass' and 'kept_dead' mistakes by being conservative
  with low-confidence predictions. The bypass logic also adopts the same
  confidence-adjusted RDP calculations for consistency.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -16..15, effectively -11..11 with current rules
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Use confidence-adjusted RDP for bypass decisions, aligning with successful designs.
    # Low confidence reduces the effective RDP, making lines less likely to bypass.
    - deconfident_rdp_gt: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, threshold: 73, scale: 4}
    - deconfident_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 4}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf              # Renamed from bypass_confidence for consistency with deconfident_ words
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO                # 0 means high confidence, 3 means low confidence

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: rdp_conf, index: signature, op: decrement} # Match means good, so increase confidence (decrease rdp_conf value)
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: increment} # Old entry not matched, decrease confidence (increase rdp_conf value)
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp_conf, index: signature, op: increment} # Replaced entry not matched, decrease confidence (increase rdp_conf value)
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Use the new deconfident ETR calculation for setting the line's eviction rank
    - set:
        etr:
          deconfident_etr_val:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 4
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: deconfident_etr_val
  place: value
  params:
    rdp_table: table
    conf_table: table
    index: signature
    div: {int: [1, 128]}
    scale: {int: [0, 16]}
    max_etr: {int: [0, 15]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is read as its
    init value (0). The maximum value for confidence is 3.
    Calculates an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale).
    If the result of the subtraction is negative, it is treated as 0.
    Calculates base_etr = adjusted_rdp / div using integer division.
    The final returned value is min(max_etr, base_etr).
- name: deconfident_exceeds_victim
  place: condition
  params:

    div:
      int: [1, 128]
    index: signature
    rdp_table: table
    scale:
      int: [0, 16]
  means: 'Reads ''rdp_val'' from rdp_table[index] and ''conf_val'' from conf_table[index]. An absent rdp_val
    is treated as 0. The conf_table is assumed to be fully initialized, and its maximum value is 3. Calculates
    an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result of the subtraction

    using integer division. Returns true if predicted_etr is greater than the absolute value of the victim
    way''s key.'
