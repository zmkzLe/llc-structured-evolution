name: unified_confidence_boost
doc: >
  Revises the ETR calculation and bypass logic to use a "confidence-boosted" approach.
  A new value word, `confidence_boosted_etr`, calculates ETR by *adding* a penalty
  for low confidence to the base RDP-derived ETR, making low-confidence lines more
  prone to eviction. This unified ETR is used for both setting line ranks and for
  bypass decisions via `incoming_boosted_etr_exceeds_victim`. The sampler's confidence
  training is also corrected to align with this interpretation of confidence (higher
  `conf_val` means higher confidence). This aims to reduce both `wrong_bypass`
  and `kept_dead` errors by consistent, confidence-aware ranking.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Use the new confidence-boosted ETR for bypass decisions
    - incoming_boosted_etr_exceeds_victim:
        rdp_table: rdp
        conf_table: bypass_confidence
        index: pc_sig
        div: 8
        scale: 2              # Consistent scale with ETR calculation
        max_etr: 11
        rdp_inf_threshold: 73

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: bypass_confidence, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: bypass_confidence, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: decrement} # Correct: decrease confidence for long age
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: bypass_confidence, index: signature, op: decrement} # Correct: decrease confidence for replaced
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Aggressively evict lines with zero confidence (low conf_val), a proven heuristic
    - when: {table: {table: bypass_confidence, index: pc_sig, cmp: eq, threshold: 0}}
      set: {etr: {const: 11}} # Set to max ETR for immediate eviction
      stop: true # Stop further ETR calculation for this line
    - set:
        etr:
          confidence_boosted_etr:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            div: 8
            scale: 2              # Consistent scale
            max_etr: 11
            rdp_inf_threshold: 73

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confidence_boosted_etr
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
    rdp_inf_threshold: {int: [0, 127]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If rdp_val is greater than rdp_inf_threshold, the returned value is max_etr.
    Otherwise, calculate base_etr = rdp_val / div using integer division.
    Calculate penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).

- name: incoming_boosted_etr_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
    rdp_inf_threshold: {int: [0, 127]}
  means: >
    Calculates an incoming ETR value for a bypass decision. It reads 'rdp_val' from
    rdp_table[index] and 'conf_val' from conf_table[index]. An absent 'rdp_val' is
    treated as 0, and an absent 'conf_val' is its init value (0). The maximum value
    for confidence is 3.
    If rdp_val is greater than rdp_inf_threshold, the incoming ETR is max_etr.
    Otherwise, calculate base_etr = rdp_val / div (integer division).
    Calculate penalty = (3 - conf_val) * scale.
    The incoming ETR is min(max_etr, base_etr + penalty).
    This condition returns true if this calculated incoming ETR is greater than the
    absolute value of the victim way's key.

- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
