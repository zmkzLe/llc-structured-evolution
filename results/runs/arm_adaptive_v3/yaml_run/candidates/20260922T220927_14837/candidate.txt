name: mj_unified_confidence_rank
doc: >
  Unifies the ETR calculation for insertion, promotion, and bypass decisions by
  introducing a single confidence-penalized ETR rank. This rank is calculated
  based on predicted reuse distance (RDP) and a penalty for low confidence.
  This approach directly addresses the inconsistency between bypass and eviction
  decisions, aiming to reduce both `wrong_bypass` and `kept_dead` errors.
  The confidence training remains aligned with previous successful designs where
  sampler matches decrease the confidence counter (implying higher confidence),
  and non-matches increase it.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: back to Mockingjay's default
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Use the new unified ETR rank for both absolute and relative bypass conditions
    - unified_etr_gt_threshold: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, div: 8, max_etr: 11, penalty_scale: 4, threshold: 9}
    - unified_etr_exceeds_victim: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, div: 8, max_etr: 11, penalty_scale: 4}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: bypass_confidence, index: signature, op: decrement} # Confidence increases on match
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment} # Confidence decreases if not reused before sweep
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment} # Confidence decreases if not reused before replacement
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}} # Back to default aging
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}} # Back to default aging
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Use the new unified ETR rank for all ETR assignments, replacing conditional logic
    - set:
        etr:
          unified_etr_penalty:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            div: 8
            max_etr: 11
            penalty_scale: 4

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access

- name: unified_etr_penalty
  place: value
  params:
    rdp_table: table
    conf_table: table
    index: signature
    div: {int: [1, 128]}
    max_etr: {int: [0, 15]}
    penalty_scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3 (width 2).
    Calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * penalty_scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: unified_etr_gt_threshold
  place: condition
  params:
    rdp_table: table
    conf_table: table
    index: signature
    div: {int: [1, 128]}
    max_etr: {int: [0, 15]}
    penalty_scale: {int: [0, 16]}
    threshold: {int: [0, 15]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_penalty' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    base_etr = rdp_val / div. penalty = (3 - conf_val) * penalty_scale.
    incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than threshold.
- name: unified_etr_exceeds_victim
  place: condition
  params:
    rdp_table: table
    conf_table: table
    index: signature
    div: {int: [1, 128]}
    max_etr: {int: [0, 15]}
    penalty_scale: {int: [0, 16]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_penalty' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    base_etr = rdp_val / div. penalty = (3 - conf_val) * penalty_scale.
    incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.
