===== STRUCTURE_DESCRIPTION =====
name: mj_confident_rank_and_bypass
doc: >
  Introduces confidence into both eviction ranking (ETR) and bypass decisions
  to address high wrong_bypass and remaining kept_dead errors. A new value word,
  deconfident_etr_from_rdp_v2, calculates ETR from a confidence-penalized RDP,
  making eviction choices more robust. The bypass logic is also tightened: the
  relative bypass condition now uses a de-confident RDP, and the absolute bypass
  requires a higher confidence level.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - high_rdp_and_confident_bypass: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, rdp_thresh: 73, conf_thresh: 1}
    - deconfident_exceeds_victim: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, div: 8, scale: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - train: {table: bypass_confidence, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - set:
        etr:
          deconfident_etr_from_rdp_v2:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            div: 8
            scale: 8
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: high_rdp_and_confident_bypass
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, rdp_thresh: {int: [0, 127]}, conf_thresh: {int: [0, 3]}}
  means: >
    True when the entry for the index in rdp_table is present and its value is
    greater than rdp_thresh, AND the entry for the same index in conf_table has a
    value greater than conf_thresh. An absent entry in rdp_table makes the condition
    false. An absent entry in conf_table is read as its init value.
- name: deconfident_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    rdp_table: table
    scale:
      int: [0, 16]
  means: 'Reads ''rdp_val'' from rdp_table[index] and ''conf_val'' from conf_table[index]. An absent rdp_val
    is treated as 0. The conf_table is assumed to be fully initialized, and its maximum value is 3. Calculates
    an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result of the subtraction
    is negative, it is treated as 0. Calculates a predicted_etr value as adjusted_rdp divided by div,
    using integer division. Returns true if predicted_etr is greater than the absolute value of the victim
    way''s key.'
- name: deconfident_etr_from_rdp_v2
  place: value
  params:
    rdp_table: table
    conf_table: table
    index: signature
    div: {int: [1, 128]}
    scale: {int: [0, 16]}
    max_etr: {int: [0, 15]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. 'conf_val' is read from conf_table, with
    absent entries using its init value. The max confidence is 3.
    Calculates an adjusted_rdp = rdp_val - ((3 - conf_val) * scale).
    If adjusted_rdp is negative, it is treated as 0.
    The final value is min(max_etr, adjusted_rdp / div), using integer division.
