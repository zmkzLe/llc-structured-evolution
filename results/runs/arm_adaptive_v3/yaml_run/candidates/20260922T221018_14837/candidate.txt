name: mj_unified_etr_confidence_v2
doc: >
  Unifies the Estimated Time to Reuse (ETR) calculation for both insertion
  and bypass decisions using a single confidence-adjusted ETR value function.
  This directly addresses the feedback regarding conflicting criteria in bypass
  and eviction decisions, aiming to reduce `wrong_bypass` and `kept_dead` errors.
  New condition words `incoming_etr_cost_exceeds_victim` and
  `incoming_etr_cost_gt_threshold` are introduced, which use the same
  `etr_with_confidence_cost` logic as insertion. The ETR aging rate is
  increased by changing `etr_clock` to trigger every 8 accesses, to address
  `kept_dead` errors. The `scale` parameter for confidence penalty is unified
  to 4 for all ETR calculations.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3                  # Changed width for faster aging (every 8 accesses)
    init: 4                   # GRANULARITY: age faster
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Unified bypass conditions using the confidence-adjusted ETR cost
    - incoming_etr_cost_gt_threshold:
        rdp_table: rdp
        conf_table: rdp_conf
        index: pc_sig
        div: 8
        scale: 4              # Unified scale
        max_etr: 11
        threshold: 9          # Derived from previous threshold 73 / div 8
    - incoming_etr_cost_exceeds_victim:
        rdp_table: rdp
        conf_table: rdp_conf
        index: pc_sig
        div: 8
        scale: 4              # Unified scale
        max_etr: 11

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf              # Consolidated confidence table
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # Too old to be a reuse distance, sweep will detrain
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence for old lines
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement} # Decrement confidence for replaced lines
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 4}} # Trigger aging every 8 accesses
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 4}}
      set: {etr_clock: {const: 0}} # Reset etr_clock after aging
    - set: {etr_clock: {add: 1}}
    # Set ETR using the unified confidence-adjusted cost function
    - set:
        etr:
          etr_with_confidence_cost:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 4          # Unified scale
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: incoming_etr_cost_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'etr_with_confidence_cost' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val' is
    treated as 0. The maximum value for confidence is 3.
    Calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.

- name: etr_with_confidence_cost
  place: value
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
    scale:
      int: [0, 16]
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    Calculates a base_etr = rdp_val / div using integer division.
    Calculates a penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
