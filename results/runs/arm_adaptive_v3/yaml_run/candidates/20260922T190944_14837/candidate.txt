===== STRUCTURE_DESCRIPTION =====
name: mj_unified_etr
doc: >
  Unifies the bypass and insertion logic using a single confidence-penalized ETR
  metric. This metric is computed by a new value word `conf_penalized_etr` and
  used for both setting the line's ETR and for bypass decisions via the new
  condition word `incoming_etr_exceeds_victim`. This aims to coordinate eviction
  and bypass to reduce both `kept_dead` and `wrong_bypass` errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; reverted to 8 bits for accuracy
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - incoming_etr_exceeds_victim:
        rdp_table: rdp
        conf_table: rdp_conf
        index: pc_sig
        div: 8
        penalty_multiplier: 2

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Unified ETR calculation using a new value word
    - set:
        etr:
          conf_penalized_etr:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            penalty_multiplier: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: conf_penalized_etr
  place: value
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    penalty_multiplier:
      int: [0, 8]
    rdp_table: table
  means: >
    Calculates an ETR value, penalizing for low confidence.
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0, an absent 'conf_val' as 0.
    The 'rdp' table must be declared with 'unset: true'.
    The maximum value for confidence is 3.
    Calculates a base_etr = rdp_val / div (integer division).
    Calculates a confidence_penalty = (3 - conf_val) * penalty_multiplier.
    The final value is min(max_etr, base_etr + confidence_penalty).
- name: incoming_etr_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    penalty_multiplier:
      int: [0, 8]
    rdp_table: table
  means: >
    Calculates a confidence-penalized ETR for the incoming line and compares its magnitude to the victim's key.
    It reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0, an absent 'conf_val' as 0.
    The 'rdp' table must be declared with 'unset: true'.
    The maximum value for confidence is 3.
    Calculates a base_etr = rdp_val / div (integer division).
    Calculates a confidence_penalty = (3 - conf_val) * penalty_multiplier.
    The predicted_etr for the incoming line is base_etr + confidence_penalty.
    Returns true if this predicted_etr is greater than the absolute value of the victim way's key.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
