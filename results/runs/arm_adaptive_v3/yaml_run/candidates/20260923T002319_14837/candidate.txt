name: mj_accuracy_trained_confidence
doc: >
  Introduces a new training mechanism for the confidence table, as suggested by the run guidance.
  A new step word, `train_confidence_from_accuracy`, is executed on a sampler hit. It compares
  the measured reuse distance (age) with the predicted reuse distance (from the RDP table)
  and adjusts the confidence score for that signature accordingly: increasing it for accurate
  predictions and decreasing it for inaccurate ones. This directly addresses the core weakness
  of previous designs where confidence was not trained based on prediction outcomes, aiming to
  reduce both `kept_dead` and `wrong_bypass` errors by improving the quality of the confidence metric.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Use the same confidence-aware bypass from the best prior design
    - deconfident_rdp_gt: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, threshold: 73, scale: 4}
    - deconfident_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 4}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      # First, train confidence based on the accuracy of the current prediction
      - train_confidence_from_accuracy: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, margin: 16}
      # Then, train the RDP table toward the measured reuse distance and free the entry
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, scale: 1, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Use the confidence-penalized ETR from the best prior design
    - when: {table: {table: rdp_conf, index: pc_sig, cmp: eq, threshold: 0}}
      set: {etr: {const: 11}}
      stop: true
    - set:
        etr:
          etr_with_confidence_cost:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: deconfident_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    rdp_table: table
    scale:
      int: [0, 16]
  means: 'Reads ''rdp_val'' from rdp_table[index] and ''conf_val'' from conf_table[index]. An absent rdp_val
    is treated as 0. The conf_table is assumed to be fully initialized, and its maximum value is 3. Calculates
    an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result of the subtraction
    is negative, it is treated as 0. Calculates a predicted_etr value as adjusted_rdp divided by div,
    using integer division. Returns true if predicted_etr is greater than the absolute value of the victim
    way''s key.'
- name: deconfident_rdp_gt
  place: condition
  params:
    conf_table: table
    index: signature
    rdp_table: table
    scale:
      int: [0, 16]
    threshold:
      int: [0, 127]
  means: 'Reads the value ''rdp_val'' from rdp_table[index] and ''conf_val'' from conf_table[index]. An
    absent entry in rdp_table is treated as a value of 0. The conf_table is assumed to be fully initialized,
    and its maximum value is 3. Calculates an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val)
    * scale). If the result of the subtraction is negative, it is treated as 0. Returns true if this adjusted_rdp
    is greater than the threshold.

    '
- name: etr_with_confidence_cost
  place: value
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
    scale:
      int: [0, 16]
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    Calculates a base_etr = rdp_val / div using integer division.
    Calculates a penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: train_confidence_from_accuracy
  place: step
  params:
    conf_table: table
    index: signature
    margin:
      int: [0, 127]
    rdp_table: table
    sample: state
  means: >
    Reads the predicted reuse distance 'pred_rd' from rdp_table[index]. If the entry is absent, nothing is done.
    Otherwise, it reads the actual reuse distance 'actual_rd' from the 'sample' state.
    If the absolute difference between 'actual_rd' and 'pred_rd' is less than or equal to the 'margin',
    the entry in conf_table[index] is incremented (saturating).
    Otherwise, it is decremented (saturating).
