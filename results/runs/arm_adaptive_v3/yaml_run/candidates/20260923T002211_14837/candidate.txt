===== STRUCTURE_DESCRIPTION =====
name: mj_confidence_accuracy_training
doc: >
  Introduces a more direct confidence training mechanism based on prediction accuracy,
  as suggested by the run guidance. On a sampler hit, the measured reuse distance
  is compared against the prediction stored at fill time. Confidence is increased
  for accurate predictions and decreased for inaccurate ones. To stay within the
  metadata budget, the sampler size is slightly reduced by decreasing extra_index_bits
  from 4 to 3.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass if the incoming line's unified ETR rank is very high (unlikely to be reused soon)
    - incoming_rank_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11, threshold: 10}
    # Bypass if the incoming line's unified ETR rank is worse than the victim's
    - incoming_rank_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 3       # Reduced from 4 to 3 to save space for new field
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
      - {name: predicted_rd, kind: counter, width: 7, init: ZERO} # Store prediction at install time
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_rdp_and_confidence:
          rdp_table: rdp
          conf_table: rdp_conf
          index: signature
          predicted_rd: predicted_rd
          actual_rd: age
          scale: 2
          min_diff: 16
          margin_abs: 8
          margin_pct: 25
        free: true
        stop: true
      - train_rdp_and_confidence:
          rdp_table: rdp
          conf_table: rdp_conf
          index: signature
          predicted_rd: predicted_rd
          actual_rd: age
          scale: 1
          min_diff: 16
          margin_abs: 8
          margin_pct: 25
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install:
      signature: {signature: pc_sig}
      timestamp: {read: set_clock}
      predicted_rd: {table: rdp, index: pc_sig, div: 1} # Store current prediction
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Calculate ETR using the new unified rank function for all lines
    - set:
        etr:
          unified_etr_rank:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: unified_etr_rank
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If 'conf_val' is 0, the returned value is max_etr.
    Otherwise, calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: incoming_rank_gt_threshold
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
    threshold: {int: [0, 15]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than threshold.
- name: incoming_rank_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.
- name: train_rdp_and_confidence
  place: step
  params:
    rdp_table: table
    conf_table: table
    index: signature
    predicted_rd: state
    actual_rd: state
    scale: {int: [1, 2]}
    min_diff: {int: [0, 127]}
    margin_abs: {int: [0, 127]}
    margin_pct: {int: [0, 100]}
  means: >
    Performs RDP and confidence training on a sampler match.
    First, it trains the rdp_table using the 'toward' operation. The target value is min(actual_rd * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, it trains the confidence table. It calculates the absolute difference between the stored prediction and the actual reuse distance: diff = abs(predicted_rd - actual_rd). It also calculates a percentage difference using integer arithmetic: pct_diff = (diff * 100) / max(1, predicted_rd). If diff is less than or equal to margin_abs, or if pct_diff is less than or equal to margin_pct, the prediction is considered accurate and the confidence entry at the given index is incremented. Otherwise, it is decremented.
