=====STRUCTURE_DESCRIPTION=====
name: mockingjay_confidence_full_clock
doc: >
  Mockingjay with a confidence table to qualify bypass predictions, intended to
  reduce wrong_bypasses. A 2-bit confidence counter (rdp_conf) is added for
  each entry in the reuse-distance predictor (RDP) table. Confidence is
  incremented on observed reuses and decremented on evictions before reuse.
  A fill is bypassed only if the RDP prediction is high AND confident, or if
  it exceeds the victim's ETR AND is confident. Unlike a previous attempt,
  the sampler's set_clock retains its full 8-bit width to avoid degrading
  reuse distance measurements.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design - KEPT AT 8 BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # Bypass only on confident long-reuse predictions
    - table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}
      is_confident: {table: rdp_conf, index: pc_sig, threshold: 2}
    - table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}
      is_confident: {table: rdp_conf, index: pc_sig, threshold: 2}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf              # NEW TABLE FOR CONFIDENCE
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # KEPT AT 8 BITS
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      # Prefetch match: train RDP and confidence, then stop
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp_conf, index: signature, op: increment} # TRAIN CONFIDENCE
        free: true
        stop: true
      # Other match (LOAD, RFO): train RDP and confidence
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
      - train: {table: rdp_conf, index: signature, op: increment} # TRAIN CONFIDENCE
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement} # DETRAIN CONFIDENCE
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement} # DETRAIN CONFIDENCE
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
=====NEW_WORDS=====
- name: is_confident
  place: condition
  params: {index: signature, table: table, threshold: {int: [0, 3]}}
  means: >
    True when the entry of the table at the index is greater than or equal to the threshold. The
    table is expected to be fully initialized (not 'unset').
