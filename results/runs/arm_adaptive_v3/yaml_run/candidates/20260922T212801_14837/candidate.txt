===== STRUCTURE_DESCRIPTION =====
name: mj_recovery_from_4bit_clock_and_unified_rank
doc: >
  Reverts the detrimental changes of the previous attempt (4-bit set_clock and
  timestamp, 8-bit tag_bits) and restores the successful architecture of the
  top-performing design (mj_unified_etr_rank). The performance decline was likely due
  to inaccurate reuse distance prediction from the reduced clock and tag width, and
  inconsistent ETR/bypass logic. This version restores an 8-bit clock for the sampler,
  10-bit tag_bits, and consistently uses the `unified_etr_rank` function for both
  ETR assignment and bypass decisions, ensuring a coherent line ranking policy.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # Reverted to 8 bits for accuracy
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass if the incoming line's unified ETR rank is very high (unlikely to be reused soon)
    - incoming_rank_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11, threshold: 10}
    # Bypass if the incoming line's unified ETR rank is worse than the victim's
    - incoming_rank_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10              # Reverted to 10 for better accuracy
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # Reverted to 8 bits
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}} # Threshold adjusted for 8-bit age
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48} # Thresholds adjusted for 8-bit age
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48} # Thresholds adjusted for 8-bit age
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}} # Threshold adjusted for 8-bit age
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}} # Threshold adjusted for 8-bit age
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Calculate ETR using the new unified rank function for all lines
    - set:
        etr:
          unified_etr_rank:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: unified_etr_rank
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If 'conf_val' is 0, the returned value is max_etr.
    Otherwise, calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: incoming_rank_gt_threshold
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
    threshold: {int: [0, 15]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than threshold.
- name: incoming_rank_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
