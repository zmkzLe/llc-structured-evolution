name: mj_agile_deconf_bypass
doc: >
  Combines agile RDP training (min_diff=8) with confidence-adjusted bypass logic.
  Instead of requiring maximum confidence for bypass, RDP predictions are
  "de-confidented" based on a 2-bit confidence counter. This makes bypass
  decisions more nuanced, penalizing low-confidence high RDP predictions
  and aiming to reduce wrong_bypass errors while improving overall accuracy.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - deconfident_rdp_gt_v2: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, threshold: 73, scale: 4}
    - deconfident_exceeds_victim_v2: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, div: 8, scale: 4}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      # Prefetch case
      - when: {type_in: [PREFETCH]}
        train: {table: bypass_confidence, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8} # Changed min_diff from 16 to 8
        free: true
        stop: true
      # Non-prefetch case
      - train: {table: bypass_confidence, index: signature, op: decrement}
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8} # Changed min_diff from 16 to 8
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access

- name: deconfident_rdp_gt_v2
  place: condition
  params:
    conf_table: table
    index: signature
    rdp_table: table
    scale: {int: [0, 16]}
    threshold: {int: [0, 127]}
  means: >
    Reads the value 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index]. An
    absent entry in rdp_table is treated as a value of 0. The conf_table is assumed to be fully initialized,
    and its maximum value is 3. Calculates an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val)
    * scale). If the result of the subtraction is negative, it is treated as 0. Returns true if this adjusted_rdp
    is greater than the threshold.

- name: deconfident_exceeds_victim_v2
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index]. An absent rdp_val
    is treated as 0. The conf_table is assumed to be fully initialized, and its maximum value is 3. Calculates
    an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result of the subtraction
    is negative, it is treated as 0. Calculates a predicted_etr value as adjusted_rdp divided by div,
    using integer division. Returns true if predicted_etr is greater than the absolute value of the victim
    way's key.
