name: mj_accuracy_training
doc: >
  Introduces a new sampler training mechanism, `train_accuracy`, to improve prediction
  quality. On a sampler hit, it compares the predicted reuse distance with the
  actual measured distance ('age'). If the prediction is accurate (within a margin),
  the signature's confidence is increased; otherwise, it's decreased. This provides
  direct positive and negative feedback, making the confidence metric more reliable.
  The RDP table is also trained as before. This change directly addresses the guidance
  to train confidence based on prediction accuracy.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass if the incoming line's unified ETR rank is very high (unlikely to be reused soon)
    - incoming_rank_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11, threshold: 10}
    # Bypass if the incoming line's unified ETR rank is worse than the victim's
    - incoming_rank_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_accuracy: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, accuracy_margin: 16, min_rdp_diff: 16}
        free: true
        stop: true
      - train_accuracy: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, accuracy_margin: 16, min_rdp_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Calculate ETR using the unified rank function for all lines
    - set:
        etr:
          unified_etr_rank:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: unified_etr_rank
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If 'conf_val' is 0, the returned value is max_etr.
    Otherwise, calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
- name: incoming_rank_gt_threshold
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
    threshold: {int: [0, 15]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than threshold.
- name: incoming_rank_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    scale: {int: [0, 16]}
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.
- name: train_accuracy
  place: step
  params:
    rdp_table: table
    conf_table: table
    index: signature
    sample: state
    scale: {int: [1, 2]}
    accuracy_margin: {int: [0, 127]}
    min_rdp_diff: {int: [0, 127]}
  means: >
    Trains both RDP and confidence based on prediction accuracy upon a sampler match.
    Reads 'predicted_rdp' from rdp_table[index]; an absent entry is treated as 0.
    Calculates 'target_rdp' = min(sample * scale, rdp_table.max).
    If abs(predicted_rdp - target_rdp) <= accuracy_margin, the prediction is considered accurate, so it increments the entry in conf_table[index].
    Otherwise, the prediction is inaccurate, so it decrements the entry in conf_table[index].
    Finally, it trains the rdp_table[index] toward target_rdp using the 'toward' operation, but only if the absolute difference is at least min_rdp_diff.
