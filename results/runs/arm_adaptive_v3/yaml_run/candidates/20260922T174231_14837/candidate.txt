name: mj_unified_conf_rank
doc: >
  Introduces a unified, confidence-aware ranking for both incoming and resident
  lines. A new value word calculates an eviction rank (ETR) by penalizing the
  predicted reuse distance (RDP) based on a confidence score. This unified ETR
  is used for setting line ranks on hits/fills and for making bypass decisions,
  aiming to resolve the conflict between wrong bypasses and keeping dead lines.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - penalized_high_rdp_bypass: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, rdp_thresh: 73, scale: 4}
    - penalized_incoming_exceeds_victim: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, rdp_thresh: 73, div: 8, scale: 4, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: bypass_confidence, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - set:
        etr:
          penalized_etr_by_conf:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            rdp_thresh: 73
            div: 8
            scale: 4
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: penalized_etr_by_conf
  place: value
  params: {rdp_table: table, conf_table: table, index: signature, rdp_thresh: {int: [0, 127]}, div: {int: [1, 128]}, scale: {int: [0, 16]}, max_etr: {int: [0, 15]}}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. 'conf_val' is read from its table, with an absent entry read as its init value (0).
    The maximum value for confidence is 3.
    Calculates an adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result is negative, it is clamped to 0.
    If adjusted_rdp is greater than rdp_thresh, this function returns max_etr.
    Otherwise, it calculates a predicted_etr = adjusted_rdp / div, using integer division, and returns min(max_etr, predicted_etr).
- name: penalized_incoming_exceeds_victim
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, rdp_thresh: {int: [0, 127]}, div: {int: [1, 128]}, scale: {int: [0, 16]}, max_etr: {int: [0, 15]}}
  means: >
    For use in bypass_when. Computes a predicted ETR for the incoming line.
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. 'conf_val' is read from its table, with an absent entry read as its init value (0).
    The maximum value for confidence is 3.
    Calculates an adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result is negative, it is clamped to 0.
    A temporary predicted_etr is calculated: if adjusted_rdp is greater than rdp_thresh, it is max_etr; otherwise, it is adjusted_rdp / div (integer division).
    The final predicted_etr is min(max_etr, temporary predicted_etr).
    Returns true if this final predicted_etr is greater than the absolute value of the victim way's key ('etr').
- name: penalized_high_rdp_bypass
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, rdp_thresh: {int: [0, 127]}, scale: {int: [0, 16]}}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' makes the condition false. 'conf_val' is read from its table, with an absent entry read as its init value (0).
    The maximum value for confidence is 3.
    Calculates an adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result is negative, it is clamped to 0.
    Returns true if adjusted_rdp is greater than rdp_thresh.
