===== STRUCTURE_DESCRIPTION =====
name: mj_agile_rdp_unified_confidence
doc: >
  Unifies the policy around a single confidence value, representing bypass
  confidence. It combines agile RDP (min_diff=8) with confidence-aware bypass
  and a new confidence-aware ETR calculation. A new 'bypass_confidence_etr'
  word sets a line's rank using both RDP and bypass confidence, evicting lines
  with high bypass confidence sooner to reduce 'kept_dead' errors. The bypass
  logic is also confidence-aware to reduce 'wrong_bypass' errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - high_rdp_and_confident_bypass: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, rdp_thresh: 73, conf_thresh: 0}
    - deconfident_exceeds_victim: {rdp_table: rdp, conf_table: bypass_confidence, index: pc_sig, div: 8, scale: 4}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: bypass_confidence, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - set:
        etr:
          bypass_confidence_etr:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            div: 8
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: bypass_confidence_etr
  place: value
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. 'conf_val' is read from its table, with
    an absent entry read as its init value (0). The maximum value for confidence
    is 3. If rdp_val is greater than 73, the function returns max_etr. Otherwise,
    it calculates a base_etr = rdp_val / div using integer division. It calculates
    an adjustment = conf_val. The final returned value is min(max_etr, base_etr + adjustment).
- name: deconfident_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    rdp_table: table
    scale:
      int: [0, 16]
  means: 'Reads ''rdp_val'' from rdp_table[index] and ''conf_val'' from conf_table[index]. An absent rdp_val
    is treated as 0. The conf_table is assumed to be fully initialized, and its maximum value is 3. Calculates
    an adjusted rdp value: adjusted_rdp = rdp_val - ((3 - conf_val) * scale). If the result of the subtraction
    is negative, it is treated as 0. Calculates a predicted_etr value as adjusted_rdp divided by div,
    using integer division. Returns true if predicted_etr is greater than the absolute value of the victim
    way''s key.'
- name: high_rdp_and_confident_bypass
  place: condition
  params: {conf_table: table, conf_thresh: {int: [0, 3]}, index: signature, rdp_table: table, rdp_thresh: {int: [0, 127]}}
  means: >
    True when the entry for the index in rdp_table is present and its value is greater than rdp_thresh,
    AND the entry for the same index in conf_table has a value greater than conf_thresh. An absent entry
    in rdp_table makes the condition false. An absent entry in conf_table is read as its init value.
