name: mockingjay_unified_rank
doc: >
  Reverts the disastrous signature and on-hit training changes from the previous
  attempt. Introduces a unified ranking function for both bypass and insertion to
  address the conflict between wrong_bypass and kept_dead errors. Two new words,
  access_etr_logic and unified_etr_exceeds_victim, implement the same logic to
  compute a line's rank from its RDP and confidence, ensuring bypass and eviction
  decisions are made on a consistent basis.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - unified_etr_exceeds_victim:
        rdp_table: rdp
        conf_table: rdp_conf
        index: pc_sig
        div: 8
        max_etr: 11
        rdp_absent_val: 0
        rdp_thresh: 73

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}

    - set: {etr_clock: {add: 1}}
    - set:

            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            max_etr: 11
            rdp_absent_val: 0
            rdp_thresh: 73

on_hit:

    stop: true
  - run: access

on_fill:

    stop: true

===== NEW_WORDS =====
- name: access_etr_logic
  place: value
  params: {rdp_table: table, conf_table: table, index: signature, div: {int: [1, 128]}, max_etr: {int: [0, 15]}, rdp_absent_val: {int: [-15, 15]}, rdp_thresh: {int: [0, 127]}}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as a special value, and an absent 'conf_val' is treated as 0.
    The 'rdp' table must be declared with 'unset: true'.
    If 'conf_val' is 0, returns max_etr.
    If 'rdp_val' is absent (but 'conf_val' is not 0), returns rdp_absent_val.
    If 'rdp_val' is greater than rdp_thresh, returns max_etr.
    Otherwise, returns the minimum of max_etr and 'rdp_val' divided by 'div' (integer division).
- name: unified_etr_exceeds_victim
  place: condition
  params: {rdp_table: table, conf_table: table, index: signature, div: {int: [1, 128]}, max_etr: {int: [0, 15]}, rdp_absent_val: {int: [-15, 15]}, rdp_thresh: {int: [0, 127]}}
  means: >
    Computes a predicted ETR value using the same logic as the 'access_etr_logic' value word.
    It reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as a special value, and an absent 'conf_val' is treated as 0.
    The 'rdp' table must be declared with 'unset: true'.
    If 'conf_val' is 0, the predicted ETR is max_etr.
    If 'rdp_val' is absent (but 'conf_val' is not 0), the predicted ETR is rdp_absent_val.
    If 'rdp_val' is greater than rdp_thresh, the predicted ETR is max_etr.
    Otherwise, the predicted ETR is the minimum of max_etr and 'rdp_val' divided by 'div'.
    The function returns true if this predicted ETR is greater than the absolute value of the victim way's key.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
