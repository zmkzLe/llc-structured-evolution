name: mj_confidence_scaled_rank
doc: >
  Addresses the wrong_bypass/kept_dead conflict by introducing a new, more consistent
  ranking logic. A new `confidence_scaled_etr` function uses confidence to
  multiplicatively scale the predicted reuse distance (RDP), rather than additively.
  Low confidence inflates the RDP, increasing the ETR, while high confidence uses the
  RDP as-is. This prevents bypassing lines with very low RDP but low confidence. This
  single ranking logic is applied consistently to insertion, promotion, and bypass decisions.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Bypass if the incoming line's scaled ETR is very high.
    - scaled_etr_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, max_etr: 11, threshold: 10}
    # Bypass if the incoming line's scaled ETR is worse than the victim's.
    - scaled_etr_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # The update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Calculate ETR using the new confidence-scaled rank function for all lines
    - set:
        etr:
          confidence_scaled_etr:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confidence_scaled_etr
  place: value
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' or 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    Calculates a confidence-based inflation factor: inflation = (4 - conf_val).
    Calculates a scaled reuse distance: scaled_rdp = rdp_val * inflation.
    The final returned value is min(max_etr, scaled_rdp / div), using integer division.
- name: scaled_etr_gt_threshold
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
    threshold: {int: [0, 15]}
  means: >
    Calculates an estimated time to reuse (ETR) for the incoming line using the
    'confidence_scaled_etr' logic: Reads 'rdp_val' from rdp_table[index] and 'conf_val'
    from conf_table[index]. An absent 'rdp_val' or 'conf_val' is treated as 0.
    The maximum value for confidence is 3. A confidence-based inflation factor is
    calculated: inflation = (4 - conf_val). A scaled reuse distance is calculated:
    scaled_rdp = rdp_val * inflation. The incoming ETR is min(max_etr, scaled_rdp / div).
    Returns true if this incoming ETR is greater than the threshold.
- name: scaled_etr_exceeds_victim
  place: condition
  params:
    conf_table: table
    div: {int: [1, 128]}
    index: signature
    max_etr: {int: [0, 15]}
    rdp_table: table
  means: >
    Calculates an estimated time to reuse (ETR) for the incoming line using the
    'confidence_scaled_etr' logic: Reads 'rdp_val' from rdp_table[index] and 'conf_val'
    from conf_table[index]. An absent 'rdp_val' or 'conf_val' is treated as 0.
    The maximum value for confidence is 3. A confidence-based inflation factor is
    calculated: inflation = (4 - conf_val). A scaled reuse distance is calculated:
    scaled_rdp = rdp_val * inflation. The incoming ETR is min(max_etr, scaled_rdp / div).
    Returns true if this incoming ETR is greater than the absolute value of the victim way's key.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
