name: mj_eviction_penalty
doc: >
  Implements targeted confidence penalty on misprediction. A new sampler field `predicted_rdp`
  stores the RDP value at fill time. On sampler eviction (`on_replace`), a new step
  `penalize_misprediction_on_eviction` decrements confidence only if the line was
  predicted to be long-reuse but was evicted without a hit. This directly addresses the
  guidance to learn from eviction mistakes, improving prediction accuracy. State is
  reduced to stay within budget by shrinking tables and sampler ways.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - incoming_rank_gt_threshold: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11, threshold: 10}
    - incoming_rank_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, scale: 2, max_etr: 11}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 10 # Reduced from 11 for budget

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 1024 # Reduced from 2048 for budget
    width: 7
    max: 95
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 1024 # Reduced from 2048 for budget
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 4 # Reduced from 5 for budget
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 10, init: ZERO} # Reduced from 11
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
      - {name: predicted_rdp, kind: counter, width: 7, init: ZERO} # New field
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 2, min_diff: 16, conf_thresh: 48}
        free: true
        stop: true
      - train_sampler_match: {rdp_table: rdp, conf_table: rdp_conf, index: signature, sample: age, scale: 1, min_diff: 16, conf_thresh: 48}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - penalize_misprediction_on_eviction:
          conf_table: rdp_conf
          index: signature
          predicted_rdp: predicted_rdp
          rdp_table: rdp
          threshold: 48
    install:
      signature: {signature: pc_sig}
      timestamp: {read: set_clock}
      predicted_rdp: {table: rdp, index: pc_sig, div: 1} # Store prediction
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - set:
        etr:
          unified_etr_rank:
            rdp_table: rdp
            conf_table: rdp_conf
            index: pc_sig
            div: 8
            scale: 2
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: incoming_rank_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
    scale:
      int: [0, 16]
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than the absolute value of the victim way's key.

  place: condition
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
    scale:
      int: [0, 16]
    threshold:
      int: [0, 15]
  means: >
    Calculates the estimated time to reuse (ETR) for the incoming line (incoming_etr)
    using the 'unified_etr_rank' logic: Reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' or 'conf_val'
    is treated as 0. The maximum value for confidence is 3.
    If 'conf_val' is 0, incoming_etr = max_etr.
    Otherwise, base_etr = rdp_val / div. penalty = (3 - conf_val) * scale.
    Incoming_etr = min(max_etr, base_etr + penalty).
    Returns true if incoming_etr is greater than threshold.
- name: penalize_misprediction_on_eviction
  place: step
  params:
    conf_table: table
    index: signature
    predicted_rdp: state
    rdp_table: table
    threshold:
      int: [0, 127]
  means: >
    This step is run when a sampler entry is evicted. It unconditionally increments the rdp_table entry
    at the given index. It then reads the 'predicted_rdp' state value. If 'predicted_rdp' is greater than
    the 'threshold', it also decrements the conf_table entry at the same index.
- name: train_sampler_match
  place: step
  params:
    conf_table: table
    conf_thresh:
      int: [0, 127]
    index: signature
    min_diff:
      int: [0, 127]
    rdp_table: table
    sample: state
    scale:
      int: [1, 2]
  means: >
    Performs training for a sampler match based on the reuse distance 'sample' (age).
    First, it trains the rdp_table using the 'toward' operation. The target value is min(sample * scale, rdp_table.max). The entry moves one step toward the target if the difference from the target is at least min_diff.
    Second, if the 'sample' value is less than conf_thresh, it increments the conf_table entry at the same index.
- name: unified_etr_rank
  place: value
  params:
    conf_table: table
    div:
      int: [1, 128]
    index: signature
    max_etr:
      int: [0, 15]
    rdp_table: table
    scale:
      int: [0, 16]
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. An absent 'conf_val' is treated as 0.
    The maximum value for confidence is 3.
    If 'conf_val' is 0, the returned value is max_etr.
    Otherwise, calculates base_etr = rdp_val / div using integer division.
    Calculates penalty = (3 - conf_val) * scale.
    The final returned value is min(max_etr, base_etr + penalty).
