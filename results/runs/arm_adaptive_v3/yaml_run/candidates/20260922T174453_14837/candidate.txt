===== STRUCTURE_DESCRIPTION =====
name: mj_unified_rank_v1
doc: >
  Introduces a unified ranking mechanism for both insertion and bypass decisions.
  A new value word calculates an ETR for incoming lines based on both RDP and confidence,
  penalizing low-confidence predictions. A corresponding new condition word compares
  this predicted ETR against the victim's ETR to make a more robust, relative bypass
  decision. This aims to simultaneously reduce 'wrong_bypass' and 'kept_dead' errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - predicted_etr_exceeds_victim:
        rdp_table: rdp
        conf_table: bypass_confidence
        index: pc_sig
        scale: 4 # Increased scale for more penalty on low confidence
        div: 8

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: bypass_confidence, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: bypass_confidence, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: bypass_confidence, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - set:
        etr:
          confidence_adjusted_etr_v2:
            rdp_table: rdp
            conf_table: bypass_confidence
            index: pc_sig
            scale: 4 # Increased scale for more penalty on low confidence
            div: 8
            max_etr: 11

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confidence_adjusted_etr_v2
  place: value
  params:
    rdp_table: table
    conf_table: table
    index: signature
    scale: {int: [0, 16]}
    div: {int: [1, 128]}
    max_etr: {int: [0, 15]}
  means: >
    Reads 'rdp_val' from rdp_table[index] and 'conf_val' from conf_table[index].
    An absent 'rdp_val' is treated as 0. 'conf_val' is read from its table, with an
    absent entry read as its init value (0). The maximum value for confidence is 3.
    Calculates an adjusted_rdp = rdp_val - (scale * (3 - conf_val)). If adjusted_rdp
    is negative, it is treated as 0. Calculates a predicted_etr = adjusted_rdp / div.
    Returns min(predicted_etr, max_etr).
- name: predicted_etr_exceeds_victim
  place: condition
  params:
    rdp_table: table
    conf_table: table
    index: signature
    scale: {int: [0, 16]}
    div: {int: [1, 128]}
  means: >
    A condition for use inside bypass_when. It reads 'rdp_val' from rdp_table[index]
    and 'conf_val' from conf_table[index]. An absent 'rdp_val' is treated as 0.
    'conf_val' is read, with an absent entry read as its init value (0).
    The maximum value for confidence is 3. It calculates an adjusted_rdp = rdp_val -
    (scale * (3 - conf_val)). If adjusted_rdp is negative, it is treated as 0.
    It then calculates a predicted_etr = adjusted_rdp / div. The condition returns
    true if this predicted_etr is greater than the absolute value of the key of
    the victim way.
