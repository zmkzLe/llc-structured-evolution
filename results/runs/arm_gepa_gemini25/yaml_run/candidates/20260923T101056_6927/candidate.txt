===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay
doc: >
  Mockingjay with dynamic ETR aging based on set activity. A per-set saturating counter
  tracks set activity (incremented on hit, decremented on miss). The ETR decrement
  rate is adjusted dynamically: faster for active sets, slower for inactive sets.
  The RDP table size is reduced to accommodate the new state. This aims to reduce
  kept_dead and evicted_sooner mistakes by making ETR aging more responsive to
  actual set usage patterns.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: set_activity_level  # New state: tracks how active a set is
    scope: per_set
    kind: saturating
    width: 3                  # 0 to 7
    init: MID                 # Initialized to 4

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 1024             # Reduced size to make room for set_activity_level
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        # Dynamic ETR decrement based on set_activity_level
        set: {etr: {clamped_sum: {a: etr, b: {negate: {etr_decrement_rate: {activity_level_field: set_activity_level, hot_threshold: 5, cold_threshold: 2, normal_rate: 1, hot_rate: 2, cold_rate: 0}}}}, min: -11, max: 11}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - set: {set_activity_level: {add: 1}} # Increment activity on hit
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - set: {set_activity_level: {add: -1}} # Decrement activity on miss (fill)
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: etr_decrement_rate
  place: value
  params: {activity_level_field: state, hot_threshold: {int: [0, 7]}, cold_threshold: {int: [0, 7]}, normal_rate: {int: [1, 4]}, hot_rate: {int: [1, 8]}, cold_rate: {int: [0, 2]}}
  means: >
    Reads the current value of the `activity_level_field` for the access's set.
    If the value is greater than `hot_threshold`, it returns `hot_rate`.
    If the value is less than `cold_threshold`, it returns `cold_rate`.
    Otherwise, it returns `normal_rate`.
- name: negate
  place: value
  params: {value: {int: [0, 8]}}
  means: >
    Returns the negative of the input `value`. For example, if `value` is 2, it returns -2.
