===== STRUCTURE_DESCRIPTION =====
# doc: Implements a bypass feedback mechanism. A small table `bypassed_log` records
# pc_sig of bypassed lines. On a subsequent miss, if the missing line's pc_sig
# is found in `bypassed_log`, the `rdp` predictor is decremented, and the log entry is cleared.
# This directly addresses 'wrong_bypass' errors.
name: mockingjay_bypass_feedback
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing. This version adds a bypass feedback mechanism: `bypassed_log` records
  signatures of bypassed lines, and on a subsequent miss, if the missing line's signature
  is in the log, the `rdp` predictor is penalized and the log entry is cleared.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_log        # New table to log signatures of bypassed lines
    scope: global
    kind: saturating
    entries: 128              # Small table for recent bypasses (128 entries * 1 bit = 16 bytes)
    width: 1                  # Just a flag: 0=not bypassed, 1=bypassed
    init: ZERO
    unset: true               # Entries start as absent/0

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # Bypass feedback: If a line was previously bypassed (its signature is in bypassed_log)
    # and it is now being accessed (miss or hit), it was a wrong bypass.
    # Decrement the RDP predictor and clear the log entry.
    - when: {table: {table: bypassed_log, index: pc_sig, cmp: eq, threshold: 1}}
      train: {table: rdp, index: pc_sig, op: decrement} # Penalize RDP
      train: {table: bypassed_log, index: pc_sig, op: decrement} # Clear log entry (set to 0)
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # This rule applies if the fill was bypassed (way == NUM_WAY)
    train: {table: bypassed_log, index: pc_sig, op: increment} # Log the signature of the bypassed line
  - run: access
