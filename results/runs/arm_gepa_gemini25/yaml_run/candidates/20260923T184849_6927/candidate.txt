===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_feedback
doc: >
  Adds a direct feedback mechanism to learn from wrong bypasses. A new table, `bypass_tracker`, stores the PC signature of instructions that cause a bypass, indexed by a hash of the bypassed line's address. On a subsequent miss to that address, the policy identifies it as a wrong bypass and decrements the stored PC signature's entry in the main RDP predictor.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {record_bypass_on_predict: {predictor: rdp, tracker: bypass_tracker, index: pc_sig, line_addr_sig: line_sig, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: line_sig
    source: line_address
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_tracker
    scope: global
    kind: counter
    entries: 2048
    width: 11 # Must be wide enough to store pc_sig
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - penalize_wrong_bypass: {predictor: rdp, tracker: bypass_tracker, line_addr_sig: line_sig}
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: record_bypass_on_predict
  place: condition
  params: {predictor: table, tracker: table, index: signature, line_addr_sig: signature, threshold: {int: [0, 127]}}
  means: >
    Checks if the value in the predictor table at the given index (pc_sig) is greater than the threshold.
    If it is, this condition has a side effect: it writes the value of the pc_sig into the tracker table at the index given by line_addr_sig.
    After the potential side effect, it returns true if the predictor's value was greater than the threshold, and false otherwise.
    An absent predictor entry is treated as 0 and returns false.
- name: penalize_wrong_bypass
  place: step
  params: {predictor: table, tracker: table, line_addr_sig: signature}
  means: >
    Checks the tracker table at the index given by line_addr_sig. If an entry is present, it signifies a wrong bypass.
    The value of this tracker entry, which is the original pc_sig that caused the bypass, is read.
    The predictor table is then trained by decrementing the entry at that original pc_sig.
    Finally, the entry in the tracker table is invalidated to consume the signal.
