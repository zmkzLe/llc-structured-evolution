===== STRUCTURE_DESCRIPTION =====
# doc: Adds a feedback mechanism to penalize wrong bypasses. It tracks PC signatures of bypassed
# lines using a new table. A subsequent miss for a tracked signature triggers a
# penalty, reducing its prediction value in the main RDP table to make future bypasses
# for that signature less likely.

name: mockingjay_bypass_feedback
doc: >
  Modifies Mockingjay to include a feedback mechanism for wrong bypass decisions.
  A new table, `bypassed_sigs`, records the pc_sig of lines that were bypassed.
  On a subsequent miss for a line whose pc_sig is in `bypassed_sigs`, the `rdp`
  entry for that signature is decremented, and the entry in `bypassed_sigs` is
  cleared. This aims to reduce `wrong_bypass` errors by making the RDP predictor
  less aggressive for signatures that were incorrectly bypassed.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_sigs       # New table to record PC signatures of bypassed lines
    scope: global
    kind: flag                # Just a flag to indicate if a signature was bypassed
    entries: 2048             # Same size as RDP table
    width: 1
    unset: true               # Entries start absent (not bypassed)

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  # Check for wrong bypass on a miss (way == NUM_WAY, implicitly)
  - when: {counter: {counter: way, cmp: eq, threshold: NUM_WAY}} # This is a miss
    detect_and_punish_wrong_bypass:
      bypassed_table: bypassed_sigs
      predictor_table: rdp
      index: pc_sig
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  # Record bypassed signature if this fill was a bypass (way == NUM_WAY)
  - when: {counter: {counter: way, cmp: eq, threshold: NUM_WAY}}
    set: {bypassed_sigs: {const: 1}} # Mark this signature as bypassed
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: detect_and_punish_wrong_bypass
  place: step
  params: {bypassed_table: table, predictor_table: table, index: signature}
  means: >
    Checks if the entry in bypassed_table at the given index is present (i.e., this signature was previously bypassed).
    If it is present, it decrements the entry in the predictor_table at the same index, and then clears
    the entry in the bypassed_table by marking it absent. This provides negative feedback for a wrong bypass decision.
