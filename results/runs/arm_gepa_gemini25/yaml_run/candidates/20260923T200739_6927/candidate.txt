===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_feedback
doc: >
  Mockingjay, enhanced with negative feedback for wrong bypass decisions.
  A new 1-bit table, `bypassed_log`, tracks PC signatures of lines that were
  bypassed. If a subsequent fill (miss) occurs for a line whose signature is
  in the `bypassed_log`, the RDP predictor for that signature is penalized
  (decremented), and the log entry is cleared. This aims to reduce `wrong_bypass`
  errors by providing direct, accurate feedback to the bypass predictor.
  Each line holds an estimated time to reuse (ETR), counted down every 8 accesses
  to its set; the victim is the line with the largest |ETR|, an overdue (negative)
  one on ties. A predictor table keyed by the hit-or-miss PC signature gives the
  ETR on insertion, and a fill predicted to be reused after every line in the set
  is bypassed. A sampled shadow cache on 32 sets measures reuse distances with an
  8-bit per-set clock and trains the predictor one step at a time; entries older
  than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and train nothing.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_log        # New table: logs signatures of bypassed lines
    scope: global
    kind: saturating
    entries: 2048             # Same size as RDP table for corresponding signatures
    width: 1                  # 1-bit flag: 0 for not bypassed, 1 for bypassed
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # If this fill was a bypass (victim returned NUM_WAY)
    words: {log_bypass_decision: {log_table: bypassed_log, index: pc_sig}} # Log that this signature was bypassed
    stop: true # Stop processing for bypassed lines, they don't get ETR or access procedure
  - # This is a regular fill (a miss that was NOT bypassed)
    words: {penalize_and_clear_bypass_log: {log_table: bypassed_log, predictor_table: rdp, index: pc_sig}}
  - run: access # Continue with the normal access procedure for a regular fill
===== NEW_WORDS =====

- name: log_bypass_decision
  place: step
  params: {log_table: table, index: signature}
  means: >
    Increments the entry in the `log_table` at the given `index`. This marks the signature
    as having been bypassed. The table should be a saturating counter (e.g., 1-bit, from 0 to 1).
- name: penalize_and_clear_bypass_log
  place: step
  params: {log_table: table, predictor_table: table, index: signature}
  means: >
    Checks the entry in `log_table` at `index`. If it is at its maximum value (meaning
    this signature was previously bypassed), it decrements the entry in `predictor_table`
    at the same `index` (to penalize the bypass prediction) and then decrements
    the entry in `log_table` (to clear the log entry and consume the signal).
    If the `log_table` entry is not at max, no action is taken on either table.
