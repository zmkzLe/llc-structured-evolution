===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_feedback
doc: >
  Adds a feedback loop to penalize wrong bypasses, by tracking bypassed signatures and penalizing the predictor on a subsequent miss.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - words: {predict_bypass_and_record: {predictor: rdp, tracker: bypass_tracker, index: pc_sig, threshold: 73}}
    - table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_tracker
    scope: global
    kind: saturating
    entries: 2048
    width: 1
    max: 1
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {table: {table: bypass_tracker, index: pc_sig, cmp: gt, threshold: 0}}
    words: {penalize_and_clear_tracker: {predictor: rdp, tracker: bypass_tracker, index: pc_sig}}
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: predict_bypass_and_record
  place: condition
  params: {predictor: table, tracker: table, index: signature, threshold: {int: [0, 127]}}
  means: >
    Checks if the value in the predictor table at the given index is greater than the threshold.
    If it is, this condition has a side effect: it trains the tracker table at the same index by calling
    increment, marking that a bypass occurred for this signature. After the potential side effect,
    it returns true if the predictor's value was greater than the threshold, and false otherwise.
    An absent predictor entry is treated as 0 and returns false.
- name: penalize_and_clear_tracker
  place: step
  params: {predictor: table, tracker: table, index: signature}
  means: >
    This action provides negative feedback for a wrong bypass. It decrements the entry in the predictor
    table at the given index. It then also decrements the entry in the tracker table at the same index,
    to consume the feedback signal and prevent repeated penalties for the same bypass event.
