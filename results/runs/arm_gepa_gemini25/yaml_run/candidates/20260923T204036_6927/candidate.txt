===== STRUCTURE_DESCRIPTION =====
# doc: Adds a bypass feedback mechanism. A small `bypassed_log` table records
# pc_sig of bypassed lines. On a subsequent miss, if the missing line's pc_sig
# is found in `bypassed_log` and matches, the `rdp` predictor is decremented,
# and the log entry is cleared. This aims to correct 'wrong_bypass' errors.
name: mockingjay_bypass_feedback
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing. This version adds a bypass feedback mechanism: a `bypassed_log`
  table records the instruction address signature (`pc_sig`) of lines that were
  bypassed. On a subsequent cache miss for a line whose `pc_sig` is found in the
  `bypassed_log` and matches, the `rdp` predictor entry for that `pc_sig` is
  decremented, and the entry in `bypassed_log` is cleared to consume the feedback.
  This aims to correct instances of 'wrong_bypass' identified by the simulator.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_log        # New table to log bypassed signatures
    scope: global
    kind: counter
    entries: 256              # Reduced to fit within budget (256 * 11 bits = 0.344 KB)
    width: 11                 # To store pc_sig, allowing comparison
    max: 2047                 # Max value for 11 bits (2^11 - 1)
    unset: true               # Allows checking for presence and storing signature

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - when: {is_miss: {}} # Only on a miss
    words: {penalize_and_clear_bypass_log: {log_table: bypassed_log, signature: pc_sig, predictor_table: rdp}}
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # If this fill was bypassed
    words: {log_bypassed_signature: {log_table: bypassed_log, signature: pc_sig}}
    stop: true # A bypassed fill doesn't need to run the access procedure for ETR.
  - run: access
===== NEW_WORDS =====

- name: log_bypassed_signature
  place: step
  params: {log_table: table, signature: signature}
  means: >
    If the entry in `log_table` at the `signature`'s index is unset, it sets
    the entry's value to the `signature`. This records that the line associated with
    `signature` was bypassed. If the entry is already set, it remains
    unchanged, effectively preventing overwriting a pending feedback signal.
- name: penalize_and_clear_bypass_log
  place: step
  params: {log_table: table, signature: signature, predictor_table: table}
  means: >
    Checks if the entry in `log_table` at the `signature`'s index is set and
    its stored value matches `signature`. If both conditions are true, it decrements
    the entry in `predictor_table` at the `signature`'s index (clamped at 0)
    and then unsets the entry in `log_table` at the `signature`'s index,
    consuming the feedback signal.
- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True if the current fill event corresponds to a bypassed fill,
    meaning the way parameter passed to replacement_cache_fill is NUM_WAY.
    False otherwise.
- name: is_miss
  place: condition
  params: {}
  means: >
    True if the current access event corresponds to a cache miss,
    meaning the way parameter passed to update_replacement_state is NUM_WAY.
    False otherwise.
