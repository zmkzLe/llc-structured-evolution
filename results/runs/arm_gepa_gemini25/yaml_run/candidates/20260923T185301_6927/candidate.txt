===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_feedback
doc: >
  Introduces a feedback mechanism for wrong bypass decisions by tracking bypassed lines
  and penalizing the RDP on subsequent misses, while reducing `set_clock` width to
  stay within metadata limits. A new table `bypassed_line_context` stores the PC
  signature of a bypassed line. On a subsequent miss to that line, the stored PC
  signature's entry in the RDP is decremented, and the context cleared.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS; reduced from 8 bits to save metadata
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: line_sig            # New signature for line address
    source: line_addr
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_line_context # New table to track bypassed lines' PC context
    scope: global
    kind: counter
    entries: 512              # Reduced size to fit metadata budget
    width: 11                 # To store pc_sig
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO} # Adjusted to match set_clock width
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - when: {way: NUM_WAY} # This is a miss
    penalize_wrong_bypass: {tracker_table: bypassed_line_context, line_index: line_sig, predictor_table: rdp}
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # If this fill was bypassed
    record_signature_in_table: {table: bypassed_line_context, index_sig: line_sig, value_sig: pc_sig}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: line_addr
  place: SignatureSource
  params: {}
  means: The line address of the access, which is the address parameter without its 6 least significant bits.
- name: record_signature_in_table
  place: step
  params: {table: table, index_sig: signature, value_sig: signature}
  means: >
    Writes the value of value_sig (the passed signature) into the table at index_sig (the passed signature).
    If an entry already exists, it is overwritten. This is used to record context like the PC signature
    of a bypassed line.
- name: penalize_wrong_bypass
  place: step
  params: {tracker_table: table, line_index: signature, predictor_table: table}
  means: >
    If an entry exists in tracker_table at line_index, it reads the stored PC signature from that entry.
    It then decrements the entry in predictor_table at the *read PC signature's index*.
    Finally, it clears the entry in tracker_table at line_index. If no entry exists, nothing happens.
    This step handles detecting and penalizing a wrong bypass decision.
