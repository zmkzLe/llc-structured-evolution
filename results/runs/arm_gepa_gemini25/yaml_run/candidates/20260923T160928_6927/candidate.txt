===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mj_wrong_bypass_feedback_v1
doc: >
  Adds a feedback loop to correct wrong bypasses. A new sampler tracks bypassed lines. A miss
  that hits in this tracker is a "wrong bypass" and penalizes the RDP entry for the PC that
  caused it.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}
  - name: bypass_tracker
    scope: global
    sets: {mirror: 1} # A single, global tracker
    extra_index_bits: 5 # 32 sets in the sampler
    ways: 16
    tag_bits: 20
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
    victim: {key: signature, extremum: min, tie_break: last, prefer_free: true} # Evict oldest/LRU-like
    on_match:
      # This is a wrong bypass. Penalize the RDP entry for the signature that caused it.
      - train: {table: rdp, index: signature, op: decrement}
      - free: true
    install:
      # Store the PC signature that is causing the bypass
      - {signature: {signature: pc_sig}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  # On a miss, check if we wrongly bypassed this line before.
  - when: {is_miss: true}
    sample: bypass_tracker

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  # On a bypass, record the line's tag and PC signature in the tracker.
  - when: {way_is_bypass: true}
    sample: bypass_tracker
  - run: access
===== NEW_WORDS =====

- name: is_miss
  place: condition
  params: {}
  means: >
    True when the `on_access` event is for a cache miss. In the ChampSim C++
    hook `update_replacement_state`, this corresponds to the `hit` parameter
    being false.
- name: way_is_bypass
  place: condition
  params: {}
  means: >
    True when the `on_fill` event is for a bypassed fill. In the ChampSim C++
    hook `replacement_cache_fill`, this corresponds to the `way` parameter being
    equal to NUM_WAY.
