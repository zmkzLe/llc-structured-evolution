===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_adaptive_bypass
doc: >
  Modifies bypass decisions using a feedback table trained by sampler outcomes to adapt to
  `wrong_bypass` and `should_bypass` errors, while reducing timestamp width to fit budget.
  A new `bypass_feedback_table` tracks the effectiveness of bypass for each PC signature.
  When the sampler detects a reuse, it decrements the feedback counter for that PC,
  indicating that bypassing it would have been a `wrong_bypass`. When the sampler
  replaces an entry before reuse, it increments the feedback counter, indicating that
  inserting it was a `should_bypass`. The bypass decision for the main cache is then
  gated by this feedback: a line is only bypassed if the `bypass_feedback_table` for
  its PC signature indicates a high confidence in bypassing. To fit within the metadata
  budget, the `set_clock` and sampler `timestamp` field widths are reduced from 8 to 7 bits.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 7                  # TIMESTAMP_BITS reduced from 8 to 7 bits
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {rdp_threshold_and_feedback: {rdp_table: rdp, feedback_table: bypass_feedback_table, index: pc_sig, rdp_threshold: 73, feedback_cmp: gte, feedback_threshold: 4}}
    - {rdp_exceeds_victim_and_feedback: {rdp_table: rdp, feedback_table: bypass_feedback_table, index: pc_sig, div: 8, feedback_cmp: gte, feedback_threshold: 4}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_feedback_table
    scope: global
    kind: saturating
    entries: 2048
    width: 3                  # 0-7, small counter
    init: MID                 # 3 for width 3

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 7, init: ZERO} # Reduced from 8 to 7 bits
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        train: {table: bypass_feedback_table, index: signature, op: decrement} # Reuse happened, so bypassing was likely wrong. Decrease bypass confidence.
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        train: {table: bypass_feedback_table, index: signature, op: decrement} # Reuse happened, so bypassing was likely wrong. Decrease bypass confidence.
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        train: {table: bypass_feedback_table, index: signature, op: increment} # Aged out, so inserting would have been useless. Increase bypass confidence.
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: bypass_feedback_table, index: signature, op: increment} # Evicted before reuse, so inserting would have been useless. Increase bypass confidence.
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: rdp_threshold_and_feedback
  place: condition
  params: {rdp_table: table, feedback_table: table, index: signature, rdp_threshold: {int: [0, 127]}, feedback_cmp: {symbol: Comparison}, feedback_threshold: {int: [0, 7]}}
  means: >
    True when the entry of rdp_table at the index is present and its value is
    greater than rdp_threshold, AND the entry of feedback_table at the index is
    present and its value compares as feedback_cmp with feedback_threshold.
    An absent rdp_table entry counts as below rdp_threshold. An absent feedback_table
    entry is treated as its init value (MID, 3) for comparison.

- name: rdp_exceeds_victim_and_feedback
  place: condition
  params: {rdp_table: table, feedback_table: table, index: signature, div: {int: [1, 127]}, feedback_cmp: {symbol: Comparison}, feedback_threshold: {int: [0, 7]}}
  means: >
    True when the rdp_table entry at the index, divided by div (integer division),
    is greater than the absolute value of the victim key of the way the victim
    search chose, AND the entry of feedback_table at the index is present and its
    value compares as feedback_cmp with feedback_threshold. This condition is only
    valid inside victim.bypass_when. An absent rdp_table entry is treated as 0.
    An absent feedback_table entry is treated as its init value (MID, 3) for comparison.
