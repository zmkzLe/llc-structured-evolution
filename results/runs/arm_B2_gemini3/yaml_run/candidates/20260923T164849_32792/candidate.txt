===== STRUCTURE_DESCRIPTION =====
name: mockingjay_safe_bypass
doc: >
  Mockingjay optimized with safe_victim_bypass and signed_greater. safe_victim_bypass
  ensures we never bypass when the chosen victim is overdue (ETR < 0), fixing wrong
  bypasses. signed_greater fixes the ETR aging condition so that lines inserted
  at ETR = +11 correctly age down over time instead of being stuck.
  Also increases RDP table to 4096 entries (12-bit signature) and reduces sampler
  ways to 4 to stay within the 48 KB budget.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - safe_victim_bypass: {table: rdp, index: pc_sig, field: etr, div: 8, limit: 73}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 4
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {signed_greater: {field: etr, min_val: -11}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: safe_victim_bypass
  place: condition
  params:
    div:
      int: [1, 16]
    field: state
    index: signature
    limit:
      int: [0, 127]
    table: table
  means: >
    True when the entry of the table at the index is present and either its value is strictly greater
    than the limit, or both the field of the victim way chosen by find_victim is greater than or equal
    to zero and the table entry divided by div (integer division) is strictly greater than that field
    of the victim way. False otherwise. Nothing is written.
- name: signed_greater
  place: condition
  params:
    field: state
    min_val:
      int: [-16, 0]
  means: >
    True when the signed per_line field of the way being visited is strictly greater than min_val.
    False otherwise. Nothing is written.
