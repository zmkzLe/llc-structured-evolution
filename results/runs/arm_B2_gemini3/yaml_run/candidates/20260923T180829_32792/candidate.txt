===== STRUCTURE_DESCRIPTION =====
name: mockingjay
doc: >
  Mockingjay, highly optimized: Introduces margin_better_bypass to strictly bypass
  lines predicted > 73, while for others it bypasses only if the victim is live (ETR >= 0)
  and the incoming line's expected ETR is strictly greater than the victim's ETR plus a margin.
  This requires a definitive difference in predicted reuse, safely reducing wrong_bypass mistakes.
  Lines forced to insert with RDP > 73 are explicitly given ETR -11. RDP table uses 4096 entries.
  Sampler uses 32 mirrored sets and 12-bit tags to fit within the 48 KB metadata budget.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 0
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {margin_better_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, limit: 73, margin: 1}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 4
    tag_bits: 12
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 0}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: -11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: margin_better_bypass
  place: condition
  params:
    div:
      int: [1, 16]
    field: state
    index: signature
    limit:
      int: [0, 127]
    margin:
      int: [0, 15]
    table: table
  means: >
    True when the entry of the table at the index is present and either its value is strictly greater
    than the limit, or both the field of the victim way chosen by find_victim is greater than or equal
    to zero and the table entry divided by div (integer division) is strictly greater than the sum of
    that field of the victim way and the margin. False otherwise. Nothing is written.
