===== STRUCTURE_DESCRIPTION =====
name: mockingjay
doc: >
  Mockingjay with live_relative_bypass and burst-filtered sampler training.
  Addresses the dominant wrong_bypass mistakes on thrashing traces (mcf, lbm,
  GemsFDTD, milc) while safeguarding against should_bypass on streaming workloads
  (soplex, cam4). In victim.bypass_when, live_relative_bypass bypasses lines
  predicted > 73 if any live lines (ETR >= 0) exist in the set; when every way in
  the set is dead (ETR < 0), bypass is suppressed to break the 100% bypass deadlock
  and permit cache re-population. In the sampler, on_replace only increments RDP
  for entries that survived to age >= 48, preventing short-term burst conflicts in
  the 4-way shadow cache from falsely inflating RDP predictions to > 73. Retains
  4096-entry RDP table and 12-bit sampler tags within the 48 KB budget.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 3
    init: 0
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {live_relative_bypass: {table: rdp, index: pc_sig, div: 8, field: etr, limit: 73}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 12

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 4096
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 4
    tag_bits: 12
    fields:
      - {name: signature, kind: counter, width: 12, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - when: {counter: {counter: age, cmp: gte, threshold: 48}}
        train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 0}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: -11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: live_relative_bypass
  place: condition
  params:
    div:
      int: [1, 16]
    field: state
    index: signature
    limit:
      int: [0, 127]
    table: table
  means: >
    True when the entry of the table at the index is present and either:
    (1) the field of the victim way chosen by find_victim is greater than or equal to zero and
    either the table entry is strictly greater than limit or the table entry divided by div
    (integer division) is strictly greater than that field of the victim way; or
    (2) the field of the victim way is strictly less than zero, the table entry is strictly greater
    than limit, and at least one way in the access set has its field greater than or equal to zero.
    False otherwise. Nothing is written.
