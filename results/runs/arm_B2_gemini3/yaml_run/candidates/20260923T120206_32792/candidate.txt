===== STRUCTURE_DESCRIPTION =====
name: mockingjay_v2
doc: >
  Mockingjay with correct ETR aging, finer-grained RDP training, and safe bypass.
  Lines with ETR = +11 now decrement properly down to overdue (-11). RDP trains
  with min_diff 8 instead of 16 to adapt to reuse changes faster. Bypassing uses
  better_bypass to never bypass when the chosen victim is overdue (ETR < 0),
  preventing wrong bypasses of streaming lines when dead lines are available.
  The unconditional bypass and insertion MAX_RD threshold is raised to 87 to
  better match the 12-way cache capacity.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - better_bypass: {table: rdp, index: pc_sig, field: etr, div: 8, limit: 87}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 8}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 8}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {signed_gt: {field: etr, min_val: -11}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 87}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: signed_gt
  place: condition
  params: {field: state, min_val: {int: [-16, 0]}}
  means: >
    True when the signed per_line field of the way being visited is strictly
    greater than min_val. False otherwise. Nothing is written.
- name: better_bypass
  place: condition
  params: {table: table, index: signature, field: state, div: {int: [1, 16]}, limit: {int: [0, 127]}}
  means: >
    True when the entry of the table at the index is present and either its
    value is strictly greater than the limit, or both the field of the victim
    way chosen by find_victim is greater than or equal to zero and the table
    entry divided by div (integer division) is strictly greater than that field
    of the victim way. False otherwise. Nothing is written.
