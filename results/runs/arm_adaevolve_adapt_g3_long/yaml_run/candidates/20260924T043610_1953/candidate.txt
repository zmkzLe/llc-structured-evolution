===== STRUCTURE_DESCRIPTION =====
name: mockingjay_demote
doc: >
  Enhance Mockingjay by demoting overdue lines twice as fast in age_set via a
  custom value word demote_overdue to reduce kept_dead on cam4, roms, and xz,
  and raising the unconditional bypass threshold from 73 to 85 to reduce
  wrong_bypass on mcf, lbm, and GemsFDTD. Preserves the MurmurHash3 finalizer
  for PC signatures to minimize aliasing.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 85}}
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 56}, table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: murmur_hash
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  age_set:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        set: {etr: {demote_overdue: {field: etr}}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 0}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: 11}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 8}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: 11}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}
===== NEW_WORDS =====

- name: murmur_hash
  place: SignatureHash
  params: {}
  means: >
    Takes the 64-bit unsigned input x. First, x = x XOR (x >> 33). Second, x = x * 0xff51afd7ed558ccd.
    Third, x = x XOR (x >> 33). Fourth, x = x * 0xc4ceb9fe1a85ec53. Finally, x = x XOR (x >> 33). All
    multiplications and shifts are 64-bit unsigned. Returns x.
- name: demote_overdue
  place: value
  params: {field: state}
  means: >
    Takes the current signed 5-bit value of the state field. If the field is
    greater than 0, returns field minus 1. If the field is less than or equal
    to 0 and greater than -10, returns field minus 2. If the field is -10 or
    less than -10, returns -11. The result is a 5-bit signed integer.
