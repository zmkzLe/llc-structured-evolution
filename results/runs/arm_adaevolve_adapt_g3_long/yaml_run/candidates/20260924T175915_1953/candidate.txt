doc: >
  Demonstrates declaring and invoking a custom SignatureHash in NEW_WORDS to
  eliminate alias conflicts in PC signatures.
state:

    scope: per_line
    kind: signed
    width: 5
    init: 0

  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: murmur_hash
    low_bits: 11
tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 0}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 8}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  place: SignatureHash
  params: {}
  means: >
