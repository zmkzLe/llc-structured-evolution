===== STRUCTURE_DESCRIPTION =====
name: mockingjay_smart_age
doc: >
  Dynamic Ensemble via Set Dueling with a novel aging algorithm.
  Introduces smart_decay in NEW_WORDS to accelerate the eviction of overdue lines.
  Lines with ETR <= 0 decay twice as fast towards -11, reducing kept_dead mistakes
  by clearing out dead lines quicker. Retains MurmurHash3 and anti-bypass guardrails.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO
  - name: psel
    scope: per_cpu
    kind: saturating
    width: 10
    init: MID

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {type_not_in: [TRANSLATION], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {type_not_in: [TRANSLATION], table: {table: rdp, index: pc_sig, cmp: gt, threshold: 56}, table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

duel:
  selector: psel
  arm_a:
    set: {etr: {table: rdp, index: pc_sig, div: 8}}
  arm_b:
    set: {etr: {table: rdp, index: pc_sig, div: 6}}
  leader_a_category: 1
  leader_b_category: 0
  leader_a_delta: 1
  leader_b_delta: -1
  follower_uses_b_when: {counter: psel, cmp: gt, threshold: MAX_OVER_2}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: murmur_hash
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  age_set:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        set: {etr: {smart_decay: {val: etr, min_val: -11}}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 0}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: 11}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 8}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: -11}}
    stop: true
  - use_duel: true
===== NEW_WORDS =====


- name: murmur_hash
  place: SignatureHash
  params: {}
  means: >
    Takes the 64-bit unsigned input x. First, x = x XOR (x >> 33).
    Second, x = x * 0xff51afd7ed558ccd. Third, x = x XOR (x >> 33).
    Fourth, x = x * 0xc4ceb9fe1a85ec53. Finally, x = x XOR (x >> 33).
    All multiplications and shifts are 64-bit unsigned. Returns x.

- name: smart_decay
  place: value
  params:
    val: state
    min_val: {int: [-15, 0]}
  means: >
    Reads the signed value of the state field val. Computes a new value:
    if the value is strictly greater than 0, the new value is the value minus 1.
    If the value is less than or equal to 0, the new value is the value minus 2.
    If the new value is strictly less than min_val, it returns min_val.
    Otherwise, it returns the new value. All arithmetic is signed.
