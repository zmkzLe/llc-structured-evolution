===== STRUCTURE_DESCRIPTION =====
name: mockingjay_ema
doc: >
  Replaces Mockingjay's slow increment/toward training with an Exponential Moving
  Average (EMA) to adapt much faster to phase changes, drastically reducing
  wrong_bypass on mcf, lbm, milc, and GemsFDTD. Dead lines now train to their
  full age (scaled by 2 on replace) immediately instead of incrementing by 1,
  eliminating the 95-miss warmup and reducing kept_dead and should_bypass on
  cam4 and xz. Bypass threshold is refined to require a minimum RDP value.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 56}, table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train_ema: {table: rdp, index: signature, val: age, scale: 2, shift: 2}
        free: true
        stop: true
      - train_ema: {table: rdp, index: signature, val: age, scale: 1, shift: 2}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train_up_ema: {table: rdp, index: signature, val: age, scale: 1, shift: 2}
        free: true
    on_replace:
      - train_up_ema: {table: rdp, index: signature, val: age, scale: 2, shift: 2}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  age_set:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 0}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: 11}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: age_set
  - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
    set: {etr: {const: 8}}
    stop: true
  - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    set: {etr: {const: 11}}
    stop: true
  - set: {etr: {table: rdp, index: pc_sig, div: 8}}
===== NEW_WORDS =====

- name: train_ema
  place: step
  params: {table: table, index: signature, val: state, scale: {int: [1, 4]}, shift: {int: [1, 4]}}
  means: >
    Reads the entry of the table at the index. Computes s as val multiplied by
    scale, clamped to the table's maximum. If the entry is absent, it is set to s.
    If present, it computes a weight w as (1 left-shifted by shift) minus 1,
    and a new value as (entry * w + s) right-shifted by shift. If the new value
    equals the entry, it sets the entry to entry + 1 if entry < s, or to entry - 1
    if entry > s. Otherwise, it sets the entry to the new value.
- name: train_up_ema
  place: step
  params: {table: table, index: signature, val: state, scale: {int: [1, 4]}, shift: {int: [1, 4]}}
  means: >
    Reads the entry of the table at the index. Computes s as val multiplied by
    scale, clamped to the table's maximum. If the entry is absent, it is set to s.
    If present and entry is less than s, it computes a weight w as (1 left-shifted
    by shift) minus 1, and a new value as (entry * w + s) right-shifted by shift;
    if the new value equals the entry, it sets the entry to entry + 1; otherwise
    it sets the entry to the new value. If present and entry is greater than or
    equal to s, it adds 1 to the entry, clamped to the table's maximum.
