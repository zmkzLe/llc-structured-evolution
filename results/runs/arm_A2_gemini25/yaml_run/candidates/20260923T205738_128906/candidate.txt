===== STRUCTURE_DESCRIPTION =====
name: mj_confident_bypass
doc: >
  Introduces a confidence mechanism to make bypass decisions more robust and reduce
  wrong bypasses. A new 2-bit confidence table is trained alongside the RDP:
  confidence is incremented on evictions of unused lines and decremented on reuse.
  Bypass is now governed by a new condition, `conf_pred_gt_relaxed_victim`, which
  only allows a bypass if the confidence in a long-reuse prediction is maximal AND
  the predicted value is worse than the victim's. This replaces Mockingjay's
  original, more aggressive bypass logic.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - conf_pred_gt_relaxed_victim:
        rdp_tbl: rdp
        conf_tbl: conf
        index: pc_sig
        div: 8
        conf_thresh: 3
        margin: 0

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: 0

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        run: train_on_match_prefetch
        free: true
        stop: true
      - run: train_on_match_normal
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        run: train_on_dead_line
        free: true
    on_replace:
      - run: train_on_dead_line
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}
  train_on_dead_line:
    - train: {table: conf, index: signature, op: increment}
    - train: {table: rdp, index: signature, op: increment}
  train_on_match_prefetch:
    - train: {table: conf, index: signature, op: decrement}
    - train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
  train_on_match_normal:
    - train: {table: conf, index: signature, op: decrement}
    - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: conf_pred_gt_relaxed_victim
  place: condition
  params:
    rdp_tbl: table
    conf_tbl: table
    index: signature
    div: {int: [1, 1024]}
    conf_thresh: {int: [0, 3]}
    margin: {int: [0, 127]}
  means: >
    Checks if the entry for the index in the confidence table 'conf_tbl' is present
    and its value is greater than or equal to 'conf_thresh'. If not, the condition
    is false. If it is, it calculates a predicted ETR for the incoming line as
    floor(rdp_tbl[index] / div). It reads the absolute value of the 'etr' field of
    the victim line ('victim_etr'). A threshold is calculated as victim_etr minus
    the margin, clamped at a minimum of 0. The condition is true if the predicted ETR
    is strictly greater than this threshold. If the 'rdp_tbl' entry is absent, the
    predicted ETR is 0, and the condition is false. The 'etr' field must be a
    per_line state field. This can only be used in 'bypass_when'.
