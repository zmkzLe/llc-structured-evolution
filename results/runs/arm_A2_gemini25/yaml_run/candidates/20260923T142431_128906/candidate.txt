===== STRUCTURE_DESCRIPTION =====
name: mj_confident_bypass_fixed
doc: >
  Fixes syntax errors in a previous attempt to implement a confidence-based bypass
  mechanism. Adds a 2-bit confidence counter (rdp_conf) table, trained by the sampler.
  Both static (long reuse) and dynamic (worse than victim) bypass decisions are now
  gated by maximum confidence. This aims to reduce wrong_bypass errors by only
  bypassing when the predictor is highly confident.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # Bypass only on high-confidence predictions
    - confident_long_reuse: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, threshold: 73}
    - confident_exceeds_victim: {rdp_table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: 1                   # Start with moderate confidence

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: rdp_conf, index: signature, op: increment}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_long_reuse
  place: condition
  params:
    conf_table: table
    index: signature
    rdp_table: table
    threshold:
      int: [0, 127]
  means: >
    True if the rdp_table entry at index is present and its value is greater than the threshold,
    AND the conf_table entry at the same index is at its maximum value. False otherwise. The rdp_table
    must be 'unset'.
- name: confident_exceeds_victim
  place: condition
  params:
    conf_table: table
    div:
      int: [1, 1024]
    index: signature
    rdp_table: table
  means: >
    True only when the entry for the index in the conf_table is at its maximum value.
    If so, it then checks if the predicted ETR for the incoming line is greater than the ETR of the victim line.
    The predicted ETR is calculated as floor(rdp_table[index] / div).
    The victim's ETR is the absolute value of the 'etr' field of the way chosen for eviction.
    The condition is true if confidence is max AND predicted ETR > victim ETR.
    False if confidence is not max, or if the rdp_table entry is absent. The conf_table entry is never absent.
