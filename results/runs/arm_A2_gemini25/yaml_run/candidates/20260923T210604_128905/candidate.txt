===== STRUCTURE_DESCRIPTION =====
name: mj_confident_victim_bypass
doc: >
  Introduces a confidence-based bypass mechanism to address conflicting bypass errors.
  This design adds a 2-bit confidence table, trained by the sampler on reuse and
  eviction events. A new bypass condition, 'confident_pred_worse_than_victim',
  only permits a bypass if the prediction is confident (confidence >= 2) AND the
  incoming line's predicted value is worse than the victim's. This is paired with
  the static bypass for very long-lived lines, aiming to reduce both wrong_bypass
  and kept_dead mistakes by making more informed decisions.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - confident_pred_worse_than_victim: {rdp_tbl: rdp, conf_tbl: conf, index: pc_sig, div: 8, conf_thresh: 2}
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: 1                   # confidence: 0..3

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: conf, index: signature, op: increment}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: conf, index: signature, op: decrement}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: conf, index: signature, op: decrement}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_pred_worse_than_victim
  place: condition
  params:
    rdp_tbl: table
    conf_tbl: table
    index: signature
    div: {int: [1, 1024]}
    conf_thresh: {int: [0, 3]}
  means: >
    Calculates a predicted ETR for the incoming line as floor(rdp_tbl[index] / div).
    Reads the absolute value of the 'etr' field of the victim line ('victim_etr').
    Reads the confidence value from conf_tbl[index].
    The condition is true if the confidence is greater than or equal to conf_thresh AND
    the predicted ETR is strictly greater than victim_etr.
    If the rdp_tbl or conf_tbl entry is absent, the condition is false.
    The 'etr' field must be a per_line state field. This can only be used in 'bypass_when'.
