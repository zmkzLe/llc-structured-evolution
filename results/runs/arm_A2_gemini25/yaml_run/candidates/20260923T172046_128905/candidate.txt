===== STRUCTURE_DESCRIPTION =====
name: mj_victim_weighted_bypass
doc: >
  Refines the bypass decision by using a victim-weighted blend for comparison.
  Instead of an equal blend of victim and average ETRs, this design gives the
  victim's ETR twice the weight of the set's average ETR when computing the
  bypass threshold. This aims for a more nuanced, direct comparison to the
  victim line, making the bypass decision more sensitive to the actual line
  being evicted. This should help reduce both wrong_bypass and kept_dead errors
  by more accurately identifying truly undesirable incoming lines.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - pred_exceeds_victim_weighted_blend: {table: rdp, index: pc_sig, div: 8, victim_weight: 2, avg_weight: 1}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: pred_exceeds_victim_weighted_blend
  place: condition
  params:
    table: table
    index: signature
    div: {int: [1, 1024]}
    victim_weight: {int: [1, 10]}
    avg_weight: {int: [1, 10]}
  means: >
    Calculates a predicted ETR for the incoming line as floor(table[index] / div).
    Calculates the integer average of the absolute values of the 'etr' field across
    all valid resident lines in the set ('avg_etr'). Finds the absolute value of the
    'etr' field of the victim line ('victim_etr').
    The condition is true if the predicted ETR is greater than the integer average of
    ('avg_weight' * 'avg_etr' + 'victim_weight' * 'victim_etr') / ('avg_weight' + 'victim_weight').
    If the table entry for the index is absent, the predicted ETR is 0, and the condition is false.
    The 'etr' field must be a per_line state field. This can only be used in 'bypass_when'.
