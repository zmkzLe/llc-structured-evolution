name: mj_conf_relaxed_victim_bypass
doc: >
  Introduces a confidence mechanism to refine bypass decisions. A new 2-bit confidence
  table, rdp_conf, is trained by the sampler: hits increase confidence, while evictions
  of unused lines decrease it. A new bypass condition, pred_gt_victim_with_conf,
  is added. It allows bypassing only when the prediction is confident (confidence is max)
  AND the incoming line's predicted ETR is greater than the victim's ETR minus a small
  margin. This 'relaxed victim' comparison, gated by high confidence, aims to reduce
  'wrong_bypass' errors by being selective, while still being aggressive enough to
  reduce 'should_bypass' errors, directly addressing the feedback from previous attempts.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    # Original Mockingjay static bypass: bypass if predicted RDP is very high
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    # New dynamic bypass: bypass if prediction is confident and ETR > relaxed victim ETR
    - pred_gt_victim_with_conf: {rdp_tbl: rdp, conf_tbl: rdp_conf, index: pc_sig, div: 8, margin: 2, conf_thresh: 3}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: 1                   # Start with moderate confidence

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - train: {table: rdp_conf, index: signature, op: increment}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp_conf, index: signature, op: decrement}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: rdp_conf, index: signature, op: decrement}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: pred_gt_victim_with_conf
  place: condition
  params:
    conf_tbl: table
    conf_thresh:
      int: [0, 3]
    div:
      int: [1, 1024]
    index: signature
    margin:
      int: [0, 127]
    rdp_tbl: table
  means: >
    Checks if the entry for the signature 'index' in table 'conf_tbl' is present and its value is
    greater than or equal to 'conf_thresh'. If not, the condition is false. If it is, it then
    calculates a predicted ETR for the incoming line as floor(rdp_tbl[index] / div). It finds the
    absolute value of the 'etr' field of the victim line ('victim_etr'). A threshold is calculated as
    victim_etr minus the margin, clamped at a minimum of 0. The condition is true if the confidence
    check passes AND the predicted ETR is strictly greater than this threshold. If the table entry
    for the index in 'rdp_tbl' is absent, the predicted ETR is 0, and the condition is false.
