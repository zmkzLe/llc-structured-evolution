===== STRUCTURE_DESCRIPTION =====
name: mj_confident_bypass
doc: >
  Introduces a confidence mechanism to moderate Mockingjay's bypass decisions. A new
  2-bit confidence table is added, indexed by PC signature. This table is trained
  by the sampler: confidence in a long reuse distance increases when a sampled line
  is evicted without being used, and decreases on a reuse hit. Bypass decisions are
  now qualified by this confidence. A line is only bypassed if its predicted reuse
  is long AND the predictor is confident. This applies to both the dynamic victim
  comparison and the static long-reuse threshold, aiming to reduce wrong bypasses
  without reintroducing the high number of kept-dead lines from previous attempts.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - confident_pred_exceeds_victim: {rdp_table: rdp, conf_table: confidence, index: pc_sig, div: 8, conf_threshold: 2}
    - confident_pred_is_long: {rdp_table: rdp, conf_table: confidence, index: pc_sig, rdp_threshold: 73, conf_threshold: 2}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
      - when: {type_in: [PREFETCH]}
        train: {table: confidence, index: signature, op: decrement}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
      - train: {table: confidence, index: signature, op: decrement}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: confidence, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
      - train: {table: confidence, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_pred_exceeds_victim
  place: condition
  params:
    conf_table: table
    conf_threshold: {int: [0, 3]}
    rdp_table: table
    index: signature
    div: {int: [1, 1024]}
  means: >
    True if the entry for the signature 'index' in 'conf_table' is present and its value is
    greater than or equal to 'conf_threshold', AND the entry for 'index' in 'rdp_table' is
    present and its value after integer division by 'div' is strictly greater than the
    absolute value of the 'etr' field of the victim line. False if either table entry
    is absent. The 'etr' field must be a per_line state field. This can only be used in 'bypass_when'.
- name: confident_pred_is_long
  place: condition
  params:
    conf_table: table
    conf_threshold: {int: [0, 3]}
    rdp_table: table
    rdp_threshold: {int: [0, 127]}
    index: signature
  means: >
    True if the entry for the signature 'index' in 'conf_table' is present and its value is
    greater than or equal to 'conf_threshold', AND the entry for 'index' in 'rdp_table' is
    present and its value is strictly greater than 'rdp_threshold'. False if either table
    entry is absent.
