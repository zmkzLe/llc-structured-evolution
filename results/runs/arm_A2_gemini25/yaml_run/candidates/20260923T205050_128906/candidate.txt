===== STRUCTURE_DESCRIPTION =====
name: mj_confident_bypass
doc: >
  Adds a confidence mechanism to Mockingjay's reuse predictor. A 2-bit confidence counter
  is associated with each RDP entry and trained in the sampler: confidence increases on
  correct long-reuse predictions and decreases on incorrect ones. A new bypass condition,
  `confident_pred_gt_victim_etr`, uses this confidence to make bypass decisions more
  robust, bypassing a line only if its predicted ETR is greater than the victim's ETR AND
  the prediction is confident. This aims to reduce both `wrong_bypass` and `kept_dead` errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - confident_pred_gt_victim_etr: {table: rdp, conf_table: rdp_conf, index: pc_sig, div: 8, conf_thresh: 2}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: rdp_conf
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {table: {table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: decrement}
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when:
          counter: {counter: age, cmp: gt, threshold: 95}
          table: {table: rdp, index: signature, cmp: gt, threshold: 73}
        train: {table: rdp_conf, index: signature, op: increment}
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - when: {table: {table: rdp, index: signature, cmp: gt, threshold: 73}}
        train: {table: rdp_conf, index: signature, op: increment}
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: confident_pred_gt_victim_etr
  place: condition
  params:
    conf_table: table
    conf_thresh:
      int: [0, 3]
    div:
      int: [1, 1024]
    index: signature
    table: table
  means: >
    Calculates a predicted ETR for the incoming line as floor(table[index] / div). An absent entry in 'table' gives a predicted ETR of 0.
    Reads the confidence value from conf_table[index]. An absent entry in 'conf_table' gives a confidence of 0.
    Finds the absolute value of the 'etr' field of the victim line ('victim_etr').
    The condition is true if the predicted ETR is greater than victim_etr AND the confidence value is greater than or equal to conf_thresh.
    The 'etr' field must be a per_line state field. This can only be used in 'bypass_when'.
