===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mj_relaxed_victim_bypass
doc: >
  Addresses the 'kept_dead' problem by making the bypass condition more lenient.
  The original policy only bypasses if an incoming line's predicted ETR is strictly
  greater than the victim's. This can lead to inserting many low-value lines that
  are only marginally better than the victim. This design introduces a new condition,
  'pred_gt_relaxed_victim_etr', which allows bypassing even if the predicted ETR is
  only close to the victim's ETR, controlled by a margin. This should increase
  bypasses of low-value lines, reducing cache pollution and 'kept_dead' mistakes.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - pred_gt_relaxed_victim_etr: {table: rdp, index: pc_sig, div: 8, margin: 1}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====
- name: pred_gt_relaxed_victim_etr
  place: condition
  params:
    table: table
    index: signature
    div: {int: [1, 1024]}
    margin: {int: [0, 127]}
  means: >
    Calculates a predicted ETR for the incoming line as floor(table[index] / div).
    Finds the absolute value of the 'etr' field of the victim line ('victim_etr').
    A threshold is calculated as victim_etr minus the margin, clamped at a minimum of 0.
    The condition is true if the predicted ETR is strictly greater than this threshold.
    If the table entry for the index is absent, the predicted ETR is 0, and the condition is false.
    The 'etr' field must be a per_line state field. This can only be used in 'bypass_when'.
