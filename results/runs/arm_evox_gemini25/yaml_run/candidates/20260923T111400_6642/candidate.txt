===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

# New state for tracking bypassed signatures
  - name: bypass_tracker_lru_clock
    scope: global
    kind: counter
    width: 32
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}
    - {record_bypassed_signature: {sampler: bypassed_signature_sampler, signature: pc_sig}} # New action for bypass_when

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  # New sampler to track bypassed signatures
  - name: bypassed_signature_sampler
    scope: global
    sets: {mirror: 1} # A small number of sets, perhaps 1 set for all signatures
    extra_index_bits: 0
    ways: 32 # 32 entries for bypassed signatures
    tag_bits: 11 # Tag is the full pc_sig
    fields:
      - {name: pc_signature_field, kind: counter, width: 11, init: ZERO}
      - {name: lru_stamp, kind: counter, width: 32, init: ZERO}
    age: {clock: bypass_tracker_lru_clock, stamp: lru_stamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - set: {lru_stamp: {tick: bypass_tracker_lru_clock}} # Update LRU on match
    on_replace:
      # If a bypassed signature is replaced, it means it was a correct bypass (not reused within the sampler's lifetime)
      - train: {table: rdp, index: pc_signature_field, op: increment} # Reinforce correct bypass
    install: {pc_signature_field: {signature: pc_sig}, lru_stamp: {tick: bypass_tracker_lru_clock}}
    after:
      - set: {bypass_tracker_lru_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access: # New rules for on_access to detect wrong bypasses
  - when: {not_a_hit: true} # This is a miss
    - when: {check_bypassed_signature: {sampler: bypassed_signature_sampler, signature: pc_sig}}
      # A miss for a signature that was previously bypassed: this was a wrong bypass
      - train: {table: rdp, index: pc_sig, op: decrement} # Make future bypass less likely
      - invalidate_bypassed_signature: {sampler: bypassed_signature_sampler, signature: pc_sig} # Remove from tracker

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
===== NEW_WORDS =====

- name: record_bypassed_signature
  place: condition
  params: {sampler: sampler, signature: signature}
  means: >
    When true, the current access's signature is installed into the specified sampler. This word itself
    does not affect the bypass decision but acts as a trigger to record the signature of the bypassed
    line. It always returns false, so it does not prevent bypass.
- name: check_bypassed_signature
  place: condition
  params: {sampler: sampler, signature: signature}
  means: >
    Returns true if an entry matching the current access's signature is found in the specified sampler.
    It does not modify the sampler.
- name: invalidate_bypassed_signature
  place: step
  params: {sampler: sampler, signature: signature}
  means: >
    Searches the specified sampler for an entry matching the current access's signature. If found,
    the entry is invalidated (marked as free).
