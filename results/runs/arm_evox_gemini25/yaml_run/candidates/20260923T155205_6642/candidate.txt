===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bhb
doc: >
  Mockingjay enhanced with a Bypass History Buffer (BHB) implemented as a small sampler
  to learn from wrong_bypass errors. When a line is bypassed, its PC signature and tag
  are recorded in the BHB. If this line is later accessed (hit or miss in the main cache),
  the BHB entry is matched, indicating a wrong_bypass. This event decrements the RDP
  table entry for that PC signature, reducing the likelihood of future bypasses for similar
  patterns. The BHB uses 2 ways and 10 tag bits per entry to stay within metadata limits.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: bhb_global_clock # New global clock for the bypass_sampler's LRU
    scope: global
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypass_sampler # New sampler to act as Bypass History Buffer (BHB)
    scope: global
    sets: {mirror: 32} # Tracks bypassed lines in sampled sets
    extra_index_bits: 4 # Matches main sampler's indexing
    ways: 2 # Small number of ways (2) to conserve metadata (128 entries * (10 tag + 11 sig + 8 ts) = 0.464 KB)
    tag_bits: 10 # Matches main sampler's tag bits for consistency
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO} # Stores pc_sig of the bypassed line
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # For LRU management
    age: {clock: bhb_global_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match: # When a line previously bypassed is accessed again
      - train: {table: rdp, index: signature, op: decrement} # Decrement RDP to penalize the wrong bypass
        free: true # Remove entry from BHB as this wrong_bypass has been "handled"
    install: {signature: {signature: pc_sig}, timestamp: {read: bhb_global_clock}}
    after:
      - set: {bhb_global_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - sample: bypass_sampler # Check the bypass_sampler on every access to detect wrong_bypasses
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: {cmp: eq, threshold: NUM_WAY}} # This fill was bypassed (way == NUM_WAY indicates bypass)
    sample: bypass_sampler # Record the bypassed line in the BHB
    stop: true # Stop further on_fill processing for bypassed lines, as it's not being installed in the main cache
  - run: access # For lines that are actually filled into the cache.
