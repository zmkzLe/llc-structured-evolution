===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_wb_correct
doc: >
  Mockingjay with wrong_bypass correction. A new bypass_tracker sampler tracks
  recently bypassed lines. On a subsequent access to a tracked line, its PC
  signature's RDP entry is decremented to discourage future bypasses. If a
  tracked line is replaced from the bypass_tracker without being re-accessed,
  its RDP entry is incremented to reinforce the bypass decision.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: bypass_clock        # Clock for the bypass_tracker sampler
    scope: global
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypass_tracker      # New sampler to track bypassed lines
    scope: global
    sets: {mirror: 4}         # A small number of sampled sets for tracking
    extra_index_bits: 0       # No extra indexing
    ways: 2                   # Fewer ways per sampler set
    tag_bits: 10              # Consistent tag bits for address matching
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO} # PC signature of the bypassed access
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # Timestamp for LRU
    age: {clock: bypass_clock, stamp: timestamp} # Uses the new global bypass_clock
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:                 # This means a previously bypassed line was accessed (wrong_bypass)
      - train: {table: rdp, index: signature, op: decrement} # Decrease RDP to discourage future bypass
      - free: true            # The line is now in cache (or hit), remove from tracker
    on_replace:               # A bypassed line was evicted from tracker without being reused
      - train: {table: rdp, index: signature, op: increment} # Increase RDP, suggests bypass was correct
    install: {signature: {signature: pc_sig}, timestamp: {read: bypass_clock}}
    after:
      - set: {bypass_clock: {add: 1}} # Increment the global bypass clock

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access: # New event to check for previously bypassed lines on every access (hit or miss)
  - sample: bypass_tracker # Check if current access is a previously bypassed line

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # Only for fills that were bypassed by the policy
    sample: bypass_tracker # Install the bypassed line's info into the tracker
    stop: true # This fill was bypassed, so don't run the normal access procedure for it
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True when the current on_fill event has the way parameter equal to NUM_WAY,
    indicating that the fill was bypassed.
