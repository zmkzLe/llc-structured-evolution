===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_history
doc: >
  Mockingjay with a bypass history table to refine bypass decisions. A new table,
  indexed by a PC signature, tracks whether lines from a given PC are typically
  bypassed (incremented on bypass) or inserted (decremented on insertion). Bypass
  decisions now require both the RDP's prediction and the bypass history table
  to allow it, aiming to reduce wrong_bypass errors for frequently useful lines.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    # Original RDP bypass condition 1, now also requires bypass_history_table to allow it
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}, bypass_history_allows: {table: bypass_history_table, index: pc_sig_bypass, threshold: 2}}
    # Original RDP bypass condition 2, now also requires bypass_history_table to allow it
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, bypass_history_allows: {table: bypass_history_table, index: pc_sig_bypass, threshold: 2}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: pc_sig_bypass      # New signature for bypass history table
    source: ip
    hash: crc3
    low_bits: 10              # Reduced bits to fit within metadata budget

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_history_table # New table to track bypass history
    scope: global
    kind: saturating
    entries: 1024             # Matches pc_sig_bypass low_bits
    width: 3                  # 0-7, allows for increment/decrement
    init: MID                 # Start neutral, 4 for 3-bit width

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # This fill was bypassed by the policy
    train: {table: bypass_history_table, index: pc_sig_bypass, op: increment} # Increment history for this PC, reinforcing bypass
    stop: true # Stop processing this bypassed fill further.
  - train: {table: bypass_history_table, index: pc_sig_bypass, op: decrement} # This fill was NOT bypassed (it was inserted)
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True when the current on_fill event has the way parameter equal to NUM_WAY,
    indicating that the fill was bypassed.
- name: bypass_history_allows
  place: condition
  params: {table: table, index: signature, threshold: {int: [0, 7]}}
  means: >
    True when the entry of the table at the index is present and its value is
    greater than or equal to the threshold; an absent entry counts as below the threshold.
