===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_learn
doc: >
  Mockingjay with bypass learning: adds a Bypass History Sampler (BHS) to track
  bypassed lines. If a bypassed line is later inserted, it signals a wrong_bypass
  and decrements its RDP entry. If a bypassed line is replaced in BHS without being
  inserted, it signals a correct_bypass and increments its RDP entry.
  A new word 'way_is_bypassed' is introduced to differentiate bypassed and inserted fills.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10              # the line address above the LLC set index
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bhs_sampler # Bypass History Sampler
    scope: global
    sets: {mirror: 4} # Reduced number of sampled sets to fit within metadata budget
    extra_index_bits: 4
    ways: 2 # Reduced number of ways to fit within metadata budget
    tag_bits: 10 # Reduced tag bits to fit within metadata budget
    fields:
      - {name: pc_signature, kind: counter, width: 11, init: ZERO} # Stores pc_sig of bypassed line
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    install: {pc_signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    on_match: # This line was previously bypassed and is now being inserted (wrong_bypass)
      - when: {counter: {counter: age, cmp: lt, threshold: 95}} # Only if the bypass was relatively recent
        train: {table: rdp, index: pc_signature, op: decrement, min_diff: 1} # Reduce RDP, encouraging future insertion
        free: true # Remove from BHS, as it's now in cache
    on_replace: # This line was bypassed, then evicted from BHS without being inserted (correct_bypass)
      - when: {counter: {counter: age, cmp: gt, threshold: 95}} # Only if it was old in BHS
        train: {table: rdp, index: pc_signature, op: increment, min_diff: 1} # Increase RDP, reinforcing bypass

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way_is_bypassed: 1} # If current fill was bypassed (way == NUM_WAY)
    sample: bhs_sampler # Record this bypass event in BHS
    stop: true # Bypassed lines do not run the rest of on_fill logic
  - when: {way_is_bypassed: 0} # If current fill was inserted (way != NUM_WAY)
    sample: bhs_sampler # Check if this inserted line was previously bypassed
  - run: access # Continue with original Mockingjay access logic for inserted lines
===== NEW_WORDS =====

- name: way_is_bypassed
  place: condition
  params: {is_bypassed: {int: [0,1]}}
  means: >
    True if the current access's way parameter is equal to NUM_WAY (indicating a bypassed fill) and is_bypassed is 1,
    or if the current access's way parameter is not equal to NUM_WAY (indicating an inserted fill) and is_bypassed is 0.
