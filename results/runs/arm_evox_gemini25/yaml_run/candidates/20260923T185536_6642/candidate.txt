===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypassed_history
doc: >
  Mockingjay with a bypassed line history. This design adds a small content-addressable
  table to track recently bypassed lines by their address tag and PC signature.
  When a fill is bypassed, its information is recorded in this history. If a later
  access (hit or miss) matches a tag in this history, it indicates a 'wrong_bypass'
  error, and the Reuse Distance Predictor (RDP) for that PC signature is decremented
  to learn from the misprediction. A global 8-bit access counter is used for LRU
  replacement within this history table, ensuring the total metadata remains within budget.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_access_counter # New: global clock for bypassed_tag_history LRU
    scope: global
    kind: counter
    width: 8                  # Small width to fit metadata budget
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

# New top-level structure to track bypassed lines
bypassed_tag_histories:
  - name: bypass_history
    scope: global
    entries: 64 # Number of entries (64 * (46 tag + 11 sig + 8 timestamp) = ~0.5KB)
    tag_bits: 46 # Line address bits to use as tag
    fields:
      - {name: sig, kind: counter, width: 11, init: ZERO} # PC signature at bypass time
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # For LRU victim
    victim: {key: timestamp, extremum: min, tie_break: first}
    on_insert:
      - set: {sig: {signature: pc_sig}, timestamp: {tick: global_access_counter}}
    on_lookup_match:
      - train: {table: rdp, index: sig, op: decrement} # This was a wrong bypass, decrement RDP
      - set: {timestamp: {tick: global_access_counter}} # Update recency

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - set: {global_access_counter: {add: 1}} # New: Increment global clock for bypass_history
  - check_bypassed_line: {history: bypass_history} # New: Check if this line was previously bypassed
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # New: if this fill was bypassed
    record_bypassed_line: {history: bypass_history} # New: record it in bypass_history
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True when the current on_fill event has the way parameter equal to NUM_WAY,
    indicating that the fill was bypassed.
- name: bypassed_tag_history
  place: top_level
  params:
    name: {type: string}
    scope: {symbol: Scope}
    entries: {int: [1, 1024]}
    tag_bits: {int: [1, 64]}
    fields: {type: list}
    victim: {type: SamplerVictim}
    on_insert: {type: list}
    on_lookup_match: {type: list}
  means: >
    A content-addressable table that stores information about recently bypassed lines
    by their line address tag. It has a fixed number of entries. When a line is
    inserted, its tag and associated fields are stored. If the table is full, an
    entry is evicted according to the victim policy. When an access occurs, if its
    tag matches an entry in this table, the on_lookup_match rules are run for the
    matching entry. The 'fields' define additional data stored with each entry.
    The table implicitly stores the tag for each entry.
- name: record_bypassed_line
  place: step
  params: {history: bypassed_tag_history}
  means: >
    Inserts the current line's tag and its associated pc_sig into the specified
    bypassed_tag_history. If the history is full, an entry is evicted based on its
    victim policy. The on_insert rules for the history are run for the new entry.
- name: check_bypassed_line
  place: step
  params: {history: bypassed_tag_history}
  means: >
    Looks up the current line's tag in the specified bypassed_tag_history. If a
    match is found, the on_lookup_match rules for the history are run for the
    matching entry.
