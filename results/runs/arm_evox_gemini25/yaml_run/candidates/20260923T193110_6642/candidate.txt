===== STRUCTURE_DESCRIPTION =====
# Mockingjay with a bypassed history sampler to correct wrong_bypass errors.
# A new sampler `bypassed_history` tracks signatures of lines that were bypassed.
# On a bypass, the line's signature is recorded in `bypassed_history`.
# On a hit or a non-bypassed fill, if the line's signature is found in `bypassed_history`,
# the RDP table is trained to decrement the predicted reuse distance for that signature,
# and the entry is removed from `bypassed_history`.
name: mockingjay_bypassed_history
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing.
  This version adds a `bypassed_history` sampler. This sampler records the PC
  signature of lines that are bypassed. If a line is later accessed (hits in the
  cache or is inserted), and its signature is found in the `bypassed_history`, it
  means it was wrongly bypassed. In this case, the RDP table entry for that signature
  is decremented (encouraging insertion next time), and the entry is removed from
  `bypassed_history`. This logic is encapsulated in new `record_bypassed_line` and
  `check_bypassed_line_reuse` step words.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_history
    scope: global
    sets: {mirror: 16} # Using a subset of cache sets for sampling
    extra_index_bits: 0
    ways: 4             # Small number of ways
    tag_bits: 11        # To store the pc_sig (11 bits)
    fields:
      - {name: sig_stored, kind: counter, width: 11, init: ZERO} # Stores the signature
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # For LRU victim policy
    age: {clock: set_clock, stamp: timestamp} # Reuse main cache's set_clock
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    # on_match, on_replace, install, after rules are handled by the new step words.

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access
  # On a hit, check if this line was previously bypassed.
  # If found, train RDP to decrement and free the entry from bypassed_history.
  - check_bypassed_line_reuse: {history_sampler: bypassed_history, rdp_table: rdp, signature: pc_sig}

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
  # If the fill is bypassed, record its signature in bypassed_history.
  - when: {is_bypassed_fill: {}}
    record_bypassed_line: {history_sampler: bypassed_history, signature: pc_sig, clock: set_clock}
  # If the fill is NOT bypassed (i.e., inserted), check if it was previously bypassed.
  # If found, train RDP to decrement and free the entry from bypassed_history.
  - when: {is_bypassed_fill: false}
    check_bypassed_line_reuse: {history_sampler: bypassed_history, rdp_table: rdp, signature: pc_sig}

===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: True when the current on_fill event has the way parameter equal to NUM_WAY, indicating that the
    fill was bypassed.

- name: record_bypassed_line
  place: step
  params:
    history_sampler: {sampler: sampler}
    signature: {signature: signature}
    clock: {state: state}
  means: >
    This step attempts to install the given signature into the specified history_sampler.
    It first searches for an entry matching the signature. If found, the entry's timestamp
    field (named 'timestamp' within the sampler) is updated with the current value of the clock.
    If not found, a free entry is allocated if available. If no free entry, a victim is
    chosen from the history_sampler according to its victim policy. The signature is then
    installed into the allocated or victimized entry (updating its 'sig_stored' field),
    and its 'timestamp' field is updated with the current value of the clock.
    The on_match, on_replace, and install rules declared for the history_sampler are NOT run.

- name: check_bypassed_line_reuse
  place: step
  params:
    history_sampler: {sampler: sampler}
    rdp_table: {table: table}
    signature: {signature: signature}
  means: >
    This step searches the specified history_sampler for an entry matching the given signature.
    If a matching entry is found, the rdp_table entry indexed by the matching sampler entry's
    'sig_stored' field is decremented. The matching entry is then marked as invalid (freed)
    in the history_sampler. If no matching entry is found, no action is taken.
    The on_match, on_replace, and install rules declared for the history_sampler are NOT run.
