===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_feedback
doc: >
  Mockingjay with Bypass Feedback Table: Addresses wrong_bypass errors by tracking
  signatures of recently bypassed lines in a new global feedback table. When a
  miss occurs for a signature that was recently bypassed, its bypass confidence in
  the RDP table is decreased, and the feedback table entry is also decremented.
  This aims to reduce wrong_bypass errors by learning from past mistakes.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_feedback_table # New table to track recently bypassed signatures
    scope: global
    kind: saturating
    entries: 2048             # Same size as RDP for consistent indexing
    width: 3                  # 3-bit saturating counter for bypass "confidence"
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - run: access
  - when: {way: NUM_WAY} # This rule applies only if the access was a cache miss (not found in LLC)
    when: {table: {table: bypass_feedback_table, index: pc_sig, cmp: gt, threshold: 0}} # Check if this signature was recently bypassed
    train: {table: rdp, index: pc_sig, op: decrement} # If it was a wrong_bypass, decrease RDP's bypass confidence
    train: {table: bypass_feedback_table, index: pc_sig, op: decrement} # Also decrement its own counter, as the bypass was a mistake

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # If the fill was bypassed (i.e., not installed in the cache)
    train: {table: bypass_feedback_table, index: pc_sig, op: increment} # Record that this signature was bypassed
    stop: true # Stop processing for bypassed fills, as they don't modify cache lines
  - run: access # This runs for actual fills (way != NUM_WAY)
