===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_learning
doc: >
  Mockingjay enhanced with a mechanism to learn from wrong_bypass errors.
  A new `bypass_history` sampler tracks the pc_sig of lines that were bypassed.
  If a line with a previously bypassed pc_sig is later accessed (hit or filled),
  it indicates a 'wrong_bypass'. In such cases, the 'rdp' table is decremented
  for that pc_sig, making future bypass decisions for it less likely, and the
  entry is removed from `bypass_history`.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_clock        # New: Used for LRU eviction in bypass_history sampler
    scope: global
    kind: counter
    width: 64
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypass_history      # New: Sampler to track signatures of bypassed lines
    scope: global
    sets: {mirror: 1}         # A single set, effectively a global history buffer
    ways: 64                  # 64 entries in the history buffer
    tag_bits: 1               # Dummy tag, actual signature stored in a field
    fields:
      - {name: bypassed_pc_sig, kind: counter, width: 11, init: ZERO} # Stores the pc_sig
      - {name: timestamp, kind: counter, width: 64, init: ZERO} # For LRU eviction
    age: {clock: global_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      # If pc_sig is found in bypass_history, it means a previously bypassed line is now accessed.
      # This indicates a WRONG_BYPASS. Decrement RDP for this signature.
      - train: {table: rdp, index: bypassed_pc_sig, op: decrement}
      - free: true            # Remove from history, as it's no longer 'bypassed'
    install: {bypassed_pc_sig: {signature: pc_sig}, timestamp: {tick: global_clock}}


procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

  record_bypassed_signature:  # New procedure to record a bypassed signature
    - sample: {name: bypass_history} # This will install the current pc_sig into bypass_history

  check_and_train_on_access:  # New procedure to check for wrong_bypass on fill/hit
    - sample: {name: bypass_history} # This will trigger on_match if pc_sig is found


on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: check_and_train_on_access # Check if this hit is a wrong_bypass
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # If this fill was bypassed
    - run: record_bypassed_signature # Record its pc_sig
    - stop: true # No further on_fill actions for a bypassed fill
  # If we reach here, it's a regular fill (not writeback, not bypassed)
  - run: check_and_train_on_access # Check if this fill is a wrong_bypass
  - run: access

===== NEW_WORDS =====

  - name: is_bypassed_fill
    place: condition
    params: {}
    means: >
      True when the current on_fill event is for a bypassed fill. This is determined
      by checking if the `way` parameter passed to the replacement_cache_fill hook
      is equal to NUM_WAY, which signifies a bypassed fill.
