===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypass_feedback
doc: >
  Mockingjay, enhanced to learn from wrong_bypass errors. A new global_clock and a
  small bypass_history_sampler track bypassed lines. When a previously bypassed
  line is re-accessed, the RDP table is trained: decremented if the reuse was
  'too soon' (wrong bypass), and incremented if 'late' (correct bypass).
  This addresses the wrong_bypass errors observed in previous feedback.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_clock
    scope: global
    kind: counter
    width: 64
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypass_history_sampler
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 0
    ways: 1
    tag_bits: 41 # (line_addr_bits 46 - log2(mirror_sets 32) 5 - extra_index_bits 0) = 41
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 64, init: ZERO}
    age: {clock: global_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    # on_match, on_replace, install are managed by the new word 'process_bypass_feedback'
    # to allow for specific training logic and explicit installation.

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - set: {global_clock: {add: 1}} # Increment global clock on every access
  - process_bypass_feedback:
      target_sampler: bypass_history_sampler
      rdp_table: rdp
      pc_signature: pc_sig
      current_time: global_clock
      wrong_bypass_rd_threshold: 73 # If reuse dist is less than 73, it's a wrong bypass
      is_bypassed_fill: 0 # Not a bypassed fill event from on_access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
  - when: {way: NUM_WAY} # This condition identifies a bypassed fill
    process_bypass_feedback:
      target_sampler: bypass_history_sampler
      rdp_table: rdp
      pc_signature: pc_sig
      current_time: global_clock
      wrong_bypass_rd_threshold: 73
      is_bypassed_fill: 1 # This is a bypassed fill event

===== NEW_WORDS =====

- name: process_bypass_feedback
  place: step
  params:
    target_sampler: sampler # The bypass_history_sampler instance
    rdp_table: table # The RDP table
    pc_signature: signature # Current access's PC signature (used for install)
    current_time: state # The global_clock counter
    wrong_bypass_rd_threshold: {int: [0, 127]} # Reuse dist below this is wrong
    is_bypassed_fill: {int: [0, 1]} # 1 if current event is a bypassed fill, 0 otherwise
  means: >
    This step manages feedback for bypass decisions.
    First, it attempts to find an entry in target_sampler matching the current line address.
    If a match is found (meaning a previously bypassed line is being reaccessed):
      It calculates the elapsed time: current_time - matched_entry.timestamp.
      If this elapsed time is less than wrong_bypass_rd_threshold, it means a wrong bypass
      occurred. The rdp_table entry for the matched_entry.signature is decremented by 1
      (saturating at 0).
      Otherwise (a correct bypass), the rdp_table entry for matched_entry.signature is
      incremented by 1 (saturating at MAX).
      The matched entry in target_sampler is then freed.
    Second, if is_bypassed_fill is 1:
      A new entry is installed in target_sampler for the current line address,
      using pc_signature as its signature and current_time as its timestamp.
      If the sampler set is full, target_sampler's victim policy is used to evict an old entry.
