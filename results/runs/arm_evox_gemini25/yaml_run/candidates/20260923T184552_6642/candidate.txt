===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_feedback
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing.

  This version adds a bypass feedback mechanism. A new signature, pc_sig_base,
  captures the IP and prefetch bit, but not the hit bit, allowing tracking of a PC
  regardless of hit/miss status. A new table, bypass_misprediction_counter, stores
  how often a given pc_sig_base has been involved in a wrong bypass. When a line is
  bypassed, this counter is incremented for its pc_sig_base. When a line hits, if
  its pc_sig_base had a non-zero entry in the counter, it means a previous bypass
  for this PC was likely a misprediction, so the counter is decremented. The bypass
  decision itself is now conditioned: a line is bypassed only if the RDP predicts a
  long reuse distance AND the bypass_misprediction_counter for that PC_base is low
  (meaning bypasses for this PC are generally reliable).

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    # Original RDP high prediction, now conditioned by bypass_misprediction_counter
    - conditional_bypass_rdp_threshold:
        rdp_table: rdp
        rdp_index: pc_sig
        rdp_cmp: gt
        rdp_threshold: 73
        mispred_table: bypass_misprediction_counter
        mispred_index: pc_sig_base
        mispred_cmp: lte # Bypass only if misprediction counter is low (0 or 1)
        mispred_threshold: 1 # Threshold 1 means bypass if counter is 0 or 1. If 2 or 3, don't bypass.
    # Original table_exceeds_victim condition remains as an OR condition
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: pc_sig_base # New signature, without hit bit, for bypass feedback
    source: ip
    fold: [prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_misprediction_counter # New table to track bypass reliability per PC_base
    scope: global
    kind: saturating
    entries: 512              # Smaller table, focused on problematic PCs
    width: 2                  # 0-3 range, 0 means RDP is often wrong for this PC_base, 3 means RDP is often right
    init: ZERO                # Start at 0, meaning no mispredictions yet
    max: 3

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  # When a line hits, if its pc_sig_base had a non-zero entry in the misprediction counter,
  # it means a previous bypass for this PC was likely a misprediction. Decrement the counter.
  - when: {table: {table: bypass_misprediction_counter, index: pc_sig_base, cmp: gt, threshold: ZERO}}
    train: {table: bypass_misprediction_counter, index: pc_sig_base, op: decrement}
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  # When a line is bypassed, increment the misprediction counter for its pc_sig_base.
  # This registers that for this PC, a bypass was attempted.
  - when: {is_bypassed_fill: {}}
    train: {table: bypass_misprediction_counter, index: pc_sig_base, op: increment}
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: True when the current on_fill event has the way parameter equal to NUM_WAY, indicating that the fill was bypassed.
- name: conditional_bypass_rdp_threshold
  place: condition
  params: {rdp_table: table, rdp_index: signature, rdp_cmp: {symbol: Comparison}, rdp_threshold: {int: [0, 127]}, mispred_table: table, mispred_index: signature, mispred_cmp: {symbol: Comparison}, mispred_threshold: {int: [0, 3]}}
  means: >
    True when rdp_table[rdp_index] satisfies rdp_cmp with rdp_threshold AND
    mispred_table[mispred_index] satisfies mispred_cmp with mispred_threshold.
    If an entry is absent from rdp_table or mispred_table, the condition for that table is false.
