===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_learning
doc: >
  Mockingjay with bypass learning: adds a new shadow cache (bypassed_tracker_sampler)
  to record information about lines that were bypassed. When a previously bypassed
  line is accessed again, and its original RDP prediction caused the bypass,
  the RDP table for that PC signature is decremented, learning from the "wrong_bypass"
  error. This aims to improve the bypass decision logic.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_tracker_sampler
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO} # PC_SIG at bypass
      - {name: rdp_at_bypass, kind: saturating, width: 7, init: ZERO, max: 95} # RDP value at bypass
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    install: {} # Installation is handled by the record_bypass_event new word
    on_match: [] # Match handling is performed by the check_bypassed_reuse new word

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - check_bypassed_reuse: {target_sampler: bypassed_tracker_sampler, rdp_table: rdp, rdp_threshold_for_bypass: 73}
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: {cmp: eq, threshold: NUM_WAY}} # This condition means the fill was bypassed
    record_bypass_event: {target_sampler: bypassed_tracker_sampler, rdp_table: rdp, pc_signature: pc_sig, set_clock_for_timestamp: set_clock}
  - run: access
===== NEW_WORDS =====

- name: record_bypass_event
  place: step
  params:
    target_sampler: {sampler: name}
    rdp_table: {table: name}
    pc_signature: {signature: name}
    set_clock_for_timestamp: {state: name}
  means: >
    Installs a new entry into `target_sampler`. If `target_sampler` is full in the relevant set, it replaces an existing entry according to `target_sampler`'s victim policy, running `on_replace` if applicable. The new entry's `signature` field is set to the current `pc_signature`. Its `rdp_at_bypass` field is set to the current value of `rdp_table` indexed by `pc_signature`. Its `timestamp` field is set to the current value of `set_clock_for_timestamp`.
- name: check_bypassed_reuse
  place: step
  params:
    target_sampler: {sampler: name}
    rdp_table: {table: name}
    rdp_threshold_for_bypass: {int: [0, 127]}
  means: >
    Searches `target_sampler` for a matching tag for the current access. If a match is found:
    Reads the `rdp_at_bypass` value from the matched sampler entry.
    If `rdp_at_bypass` is greater than `rdp_threshold_for_bypass` (Mockingjay's MAX_RD, 73),
    it trains `rdp_table` at the `signature` stored in the matched sampler entry by decrementing.
    After training, it frees the matched entry from `target_sampler`. If no match is found, no action is taken.
