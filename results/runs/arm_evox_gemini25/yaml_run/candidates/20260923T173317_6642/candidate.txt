===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bht
doc: >
  Adds a bypass history table and a global clock, using a new word `learn_from_bypass`
  to record bypassed lines and decrement the RDP for lines that were wrongly bypassed.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_clock
    scope: global
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_history_table
    scope: global
    kind: counter
    entries: 1024             # Tracks 1024 recently bypassed PC signatures
    width: 8                  # Stores global_clock timestamp
    init: ZERO
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - set: {global_clock: {add: 1}} # Increment global clock on every access
  - learn_from_bypass:
      history_table: bypass_history_table
      history_index_sig: pc_sig
      rdp_table: rdp
      current_clock: global_clock
      max_age: 32 # Remember bypassed lines for 32 global clock cycles
      train_delta: -1 # Decrement RDP to encourage insertion if wrongly bypassed
      action: check_access
  - sample: sampled_cache

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {counter: {counter: way, cmp: eq, threshold: NUM_WAY}} # Check if the fill was bypassed
    learn_from_bypass:
      history_table: bypass_history_table
      history_index_sig: pc_sig
      rdp_table: rdp # Not used for 'record_bypass' action, but required by params
      current_clock: global_clock
      max_age: 0 # Not used for 'record_bypass' action
      train_delta: 0 # Not used for 'record_bypass' action
      action: record_bypass
  - run: access
===== NEW_WORDS =====

- name: learn_from_bypass
  place: step
  params:
    history_table: {symbol: Table}
    history_index_sig: {symbol: Signature}
    rdp_table: {symbol: Table}
    current_clock: {state: state}
    max_age: {int: [0, 255]}
    train_delta: {int: [-10, 10]}
    action: {symbol: BypassLearningAction}
  means: >
    Performs bypass learning. If action is 'record_bypass', it writes the current_clock
    value into history_table at history_index_sig, marking it present.
    If action is 'check_access', it reads the entry from history_table at history_index_sig.
    If the entry is present and its age (computed as (current_clock - stored_timestamp + 256) % 256,
    where stored_timestamp is the value from history_table) is less than or equal
    to max_age, it means a previously bypassed line was accessed. In this case,
    rdp_table at history_index_sig is trained by adding train_delta to its value,
    clamped by rdp_table's kind. The entry in history_table is then marked absent.
    If the entry is present but its age is greater than max_age, it is marked absent.
    An absent entry is ignored for 'check_access'.
- name: BypassLearningAction
  place: symbol_list
  members:
    - record_bypass
    - check_access
