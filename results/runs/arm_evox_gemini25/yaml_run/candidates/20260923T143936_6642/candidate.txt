===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_learning
doc: >
  Mockingjay with bypass learning: adds a new shadow cache (bypassed_tracker_sampler)
  to record the PC signatures of lines that were bypassed from the main LLC.
  When a subsequent miss occurs for a line whose PC signature is found in this
  tracker, it indicates a "wrong_bypass" error. The RDP table entry for that
  PC signature is then decremented, providing negative feedback to improve
  future bypass decisions. A global access counter is added to track overall
  cache activity.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_access_count # A global counter incremented on every access
    scope: global
    kind: counter
    width: 64
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_tracker_sampler # New sampler to track bypassed lines
    scope: global
    sets: {mirror: 16} # Small number of sets for tracking bypassed lines
    ways: 4 # Small number of ways per set
    tag_bits: 10 # Uses the same tag bits as other samplers
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO} # Stores the PC signature of the bypassed line
    victim: {prefer_free: true} # Simple victim policy: prefer free ways, else replaces the first way
    on_match:
      - when: {way: NUM_WAY} # This rule triggers only if the current access is a miss in the main cache
        train: {table: rdp, index: signature, op: decrement} # Decrement RDP as this was a wrong_bypass
        free: true # Remove the entry from the tracker since the line is now being handled
    install: {signature: {signature: pc_sig}} # Records the PC signature of the newly bypassed line

procedures:
  cache_access_logic:         # The original access procedure, now for lines actually in the cache
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - sample: bypassed_tracker_sampler # Check if the current access (miss or hit) was for a previously bypassed line
  - set: {global_access_count: {add: 1}} # Increment a global access counter on every tag check

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: cache_access_logic

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # If this fill was bypassed from the main cache
    sample: bypassed_tracker_sampler # Record its PC signature in the tracker
    stop: true # Stop here, do not run cache_access_logic for a line that was bypassed
  - run: cache_access_logic # Only run cache_access_logic for lines actually filled into the cache
