===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bht
doc: >
  Adds a Bypass History Table (BHT) to track recently bypassed lines. When a bypassed
  line is later accessed, the RDP entry for its PC signature is decremented to learn
  from wrong_bypass decisions. A global clock is added for timestamping.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_clock
    scope: global
    kind: counter
    width: 32                 # Increased width to allow for longer tracking
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: addr_sig
    source: ip # Using IP as a proxy for address, as line address is not directly available for signature source
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bht
    scope: global
    kind: counter
    entries: 1024             # Smaller table for recently bypassed lines
    width: 32                 # Stores global_clock timestamp
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - set: {global_clock: {add: 1}} # Increment global clock on every access
  - when: {table: {table: bht, index: addr_sig, cmp: not_absent}} # If this address was previously bypassed
    train: {table: rdp, index: pc_sig, op: decrement} # Decrement RDP to penalize wrong bypass
    set: {bht: {const: 0}} # Clear BHT entry as it's no longer bypassed (or it was a wrong bypass)

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # Only record if the fill was actually bypassed
    record_bypassed: {table: bht, index: addr_sig, timestamp: global_clock}
    stop: true # Stop processing this fill if it was bypassed and recorded
  - run: access

===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: True when the current on_fill event has the way parameter equal to NUM_WAY, indicating that the
    fill was bypassed.
- name: record_bypassed
  place: step
  params: {table: table, index: signature, timestamp: state}
  means: >
    Records the current value of 'timestamp' into the specified 'table' at the given 'index'.
    This effectively stores the time when the line was bypassed. If an entry already exists,
    its timestamp is updated.
