===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay_bypasstracker
doc: >
  Mockingjay with a bypass tracker: Reduced the main sampler's ways to free up budget.
  Added a small `bypassed_lines_sampler` to track recently bypassed lines.
  When a line that was previously bypassed is accessed (a miss), it decrements the RDP entry for its signature (wrong_bypass).
  When a bypassed line is evicted from the `bypassed_lines_sampler` without being re-accessed, it increments the RDP entry (should_bypass).
  This improves the RDP's accuracy for bypass decisions, addressing `wrong_bypass` and `should_bypass` errors.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 3                   # Reduced from 5 to free up budget for bypass tracker
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_lines_sampler # New sampler to track recently bypassed lines
    scope: global
    sets: {mirror: 32}        # Sampled sets for this tracker
    extra_index_bits: 1       # 2 sampler sets per sampled set
    ways: 2                   # Two entries per sampler set
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: used, kind: flag, width: 1, init: ZERO} # Tracks if the bypassed line was re-accessed
      - {name: timestamp, kind: counter, width: 8, init: ZERO} # For age-based victim
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      # If a previously bypassed line is accessed again (miss in LLC), it was a wrong_bypass.
      # Decrement RDP for this signature to discourage future bypasses.
      - train: {table: rdp, index: signature, op: decrement}
      - set: {used: {const: 1}} # Mark as used
      - set: {timestamp: {read: set_clock}} # Update timestamp
    on_replace:
      # If a bypassed line is evicted from this sampler without being used, it was a should_bypass.
      # Increment RDP for its signature to encourage future bypasses.
      - when: {counter: {counter: used, cmp: eq, threshold: 0}}
        train: {table: rdp, index: signature, op: increment}
    install:
      # On a bypassed fill, record the signature and mark as unused.
      {signature: {signature: pc_sig}, used: {const: 0}, timestamp: {read: set_clock}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - sample: bypassed_lines_sampler # Sample the bypass tracker on every access
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # If this is a bypassed fill (fill event with way == NUM_WAY)
    sample: bypassed_lines_sampler # Install the bypassed line's signature into the tracker
    stop: true # Stop processing for this fill, as no line was installed into the LLC
  - run: access # For non-bypassed fills, run the normal access procedure to set ETR
