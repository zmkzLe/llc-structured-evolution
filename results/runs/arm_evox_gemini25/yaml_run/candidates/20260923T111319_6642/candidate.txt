===== STRUCTURE_DESCRIPTION =====
# Doc: Implemented a new sampler, `bypassed_tracker`, and a new condition word, `was_wrongly_bypassed`, to address `wrong_bypass` errors.
# The tracker records signatures of bypassed lines, and on a subsequent miss to such a line, it trains the main RDP table
# to be less aggressive in bypassing. This is the first step in addressing the wrong_bypass feedback.
name: mockingjay_bypasstracker
doc: >
  Mockingjay with a bypass tracker: Adds a shadow cache `bypassed_tracker` to sample
  and record signatures of lines that were bypassed. A new condition word `was_wrongly_bypassed`
  checks if a current miss corresponds to a previously bypassed line. If so, it trains the
  RDP table to reduce the bypass threshold, making the policy less prone to wrong bypasses.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_access_count # Added for sampler timestamping
    scope: global
    kind: counter
    width: 64
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence   # New table to track bypass confidence
    scope: global
    kind: saturating
    entries: 2048
    width: 3
    max: 7                    # 3-bit saturating counter
    init: MID                 # Start in the middle

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_tracker    # New sampler to track bypassed lines
    scope: global
    sets: {mirror: 16}        # Fewer sampled sets for bypassed lines
    extra_index_bits: 2       # Fewer extra index bits
    ways: 4
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: bypass_time, kind: counter, width: 64, init: ZERO}
    victim: {key: bypass_time, extremum: min, tie_break: first, prefer_free: true}
    on_match:                 # If a bypassed line is accessed (miss), it was a wrong bypass
      - train: {table: bypass_confidence, index: signature, op: decrement} # Reduce confidence in bypassing this signature
      - free: true            # Remove from tracker, it was re-accessed
    on_replace:
      - when: {counter: {counter: bypass_confidence, cmp: lt, threshold: MID}} # If confidence is low, means previous bypass was bad
        train: {table: rdp, index: signature, op: decrement} # Adjust RDP to be less aggressive for this signature
    install: {signature: {signature: pc_sig}, bypass_time: {tick: global_access_count}} # Record signature and time of bypass

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - set: {global_access_count: {add: 1}} # Increment global access counter

    # Check if this miss was a wrong bypass
    - when: {was_wrongly_bypassed: {sampler: bypassed_tracker, index: pc_sig}}
      train: {table: rdp, index: pc_sig, op: decrement} # Adjust RDP to be less aggressive
      train: {table: bypass_confidence, index: pc_sig, op: decrement} # Decrease confidence for this signature

    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - run: access
  - when: {bypass_when: [{table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}, {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}]} # If the line was bypassed
    sample: bypassed_tracker # Record the bypassed line's signature
    train: {table: bypass_confidence, index: pc_sig, op: increment} # Increase confidence for this signature
  - run: access # Re-run access to ensure sampler and other updates happen after potential bypass recording
===== NEW_WORDS =====

- name: was_wrongly_bypassed
  place: condition
  params: {sampler: sampler, index: signature}
  means: >
    True when a lookup in the specified sampler for the given signature results in a match.
    This signifies that the current access (a miss) corresponds to a line that was previously
    bypassed and is now being re-accessed. The sampler's on_match rule will be triggered.
