===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_feedback
doc: >
  Reduced ETR width to 4 bits, and added a Bypass Feedback Mechanism to improve bypass decisions.
  A small shadow cache (bypass_tracker) records PC signatures of lines that were bypassed.
  On a cache miss, if the line was previously bypassed, the bypass_feedback table for its PC signature
  is decremented (indicating a 'wrong bypass'). If a bypassed line is evicted from the tracker
  without being accessed, the bypass_feedback table is incremented (indicating a 'right bypass').
  The victim selection's bypass_when logic is modified to be more aggressive for PC signatures
  with positive bypass_feedback, and less aggressive for those with negative feedback, using a new condition word.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 4                  # Reduced from 5 bits to save space for bypass tracking. Holds -7..7.
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # Combined original RDP logic with new bypass_feedback
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}, has_positive_bypass_feedback: {signature: pc_sig}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}, has_positive_bypass_feedback: {signature: pc_sig}}
    - {table: {table: bypass_feedback, index: pc_sig, cmp: eq, threshold: 3}} # Force bypass if feedback is strongly positive (3)

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_feedback   # New table to store feedback on bypass decisions
    scope: global
    kind: saturating
    entries: 256              # Small table to fit within budget
    width: 2                  # 0-3 range, 1 is MID
    init: MID

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypass_tracker      # New sampler to track bypassed lines and their PC signatures
    scope: global
    sets: category_zero       # Sampled sets (1/8 of total sets)
    extra_index_bits: 0       # No extra index bits
    ways: 2                   # 2 ways per sampled set (512 sets * 2 ways = 1024 entries)
    tag_bits: 10              # Same tag bits as main sampler
    fields:
      - {name: pc_sig_field, kind: counter, width: 11, init: ZERO} # Stores PC signature of the bypassed line
      - {name: bypassed_flag, kind: flag, width: 1, init: ZERO}    # Set to 1 if line was bypassed, 0 if accessed after bypass
      - {name: timestamp, kind: counter, width: 8, init: ZERO}     # Timestamp for age calculation
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    install: {pc_sig_field: {signature: pc_sig}, bypassed_flag: {const: 1}, timestamp: {read: set_clock}} # On install, mark as bypassed
    on_match: # If a bypassed line is accessed (main cache miss, but found in tracker)
      - when: {counter: {counter: bypassed_flag, cmp: eq, threshold: 1}} # Only if it was marked as bypassed
        train: {table: bypass_feedback, index: pc_sig_field, op: decrement} # Decrement feedback (wrong bypass)
        set: {bypassed_flag: {const: 0}, timestamp: {read: set_clock}} # Reset flag, update timestamp
    on_replace: # If a bypassed line is evicted from tracker without being accessed
      - when: {counter: {counter: bypassed_flag, cmp: eq, threshold: 1}} # Only if it was marked as bypassed
        train: {table: bypass_feedback, index: pc_sig_field, op: increment} # Increment feedback (right bypass)

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 7, abs: true}} # Adjusted threshold for 4-bit ETR
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 7}} # Adjusted INF_ETR for 4-bit ETR
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access: # New rule to track bypassed lines on main cache misses
  - when: {way: NUM_WAY} # This means it's a miss in the main cache
    sample: bypass_tracker # Check bypass_tracker, triggers on_match/on_replace logic
  - run: access

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -7}} # Adjusted -INF_ETR for 4-bit ETR
    stop: true
  - when: {way: NUM_WAY} # If this fill was bypassed (i.e., line was not inserted into LLC)
    sample: bypass_tracker # Install/update in bypass_tracker, mark as bypassed
  - run: access
===== NEW_WORDS =====

- name: has_positive_bypass_feedback
  place: condition
  params: {signature: signature}
  means: >
    True if the 'bypass_feedback' table entry for the given 'signature' is greater than 0.
    This indicates that bypassing lines with this signature has historically not resulted in a 'wrong bypass' (i.e., feedback is not 0).
