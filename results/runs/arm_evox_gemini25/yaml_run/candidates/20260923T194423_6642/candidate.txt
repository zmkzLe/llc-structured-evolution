===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bf_feedback
doc: >
  Adds a Bloom filter to track bypassed signatures and trains the predictor down on a wrong bypass, to address high wrong_bypass counts.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 1024             # Reduced from 2048 to make space for the bypass_filter
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_filter
    scope: global
    kind: counter
    entries: 4096
    width: 1
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - when: { words: { query_bloom_filter: { table: bypass_filter, index: pc_sig, num_hashes: 2 } } }
      train: { table: rdp, index: pc_sig, op: decrement }
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: { words: { is_bypassed_fill: {} } }
    words: { add_to_bloom_filter: { table: bypass_filter, index: pc_sig, num_hashes: 2 } }
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True when the current on_fill event has the way parameter equal to NUM_WAY, indicating that the
    fill was bypassed.
- name: add_to_bloom_filter
  place: step
  params: {table: table, index: signature, num_hashes: {int: [1, 4]}}
  means: >
    This word implements the insertion operation for a Bloom filter.
    It computes `num_hashes` separate hash values from the given `index` signature.
    For each computed hash value, it calculates a position in the given `table`
    (hash modulo table_size) and sets the 1-bit counter at that position to 1.
    The hash functions are variants of CRC32, created by XORing the input with a small integer
    from 0 to `num_hashes`-1 before hashing.
- name: query_bloom_filter
  place: condition
  params: {table: table, index: signature, num_hashes: {int: [1, 4]}}
  means: >
    This word implements the query operation for a Bloom filter.
    It computes `num_hashes` separate hash values from the given `index` signature, using the same
    hash functions as `add_to_bloom_filter`. For each computed hash value, it calculates a position
    in the given `table` and checks the value of the 1-bit counter at that position.
    The condition returns true if and only if all checked counters are set to 1.
    It returns false otherwise.
