===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing.

  This design addresses wrong_bypass errors by tracking signatures of bypassed lines
  in a new `bypassed_tracker` table. If an incoming line's signature is in this
  tracker, the bypass is prevented, and the tracker entry is decremented. If a line
  is bypassed, its signature is added to the tracker.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    # Original bypass conditions, now modified to also check if the signature is in bypassed_tracker
    - rdp_gt_threshold_if_not_tracked: {rdp_table: rdp, rdp_index: pc_sig, rdp_threshold: 73, tracker_table: bypassed_tracker, tracker_index: pc_sig, tracker_threshold: 1}
    - rdp_exceeds_victim_if_not_tracked: {rdp_table: rdp, rdp_index: pc_sig, div: 8, tracker_table: bypassed_tracker, tracker_index: pc_sig, tracker_threshold: 1}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypassed_tracker    # New table to track signatures of mistakenly bypassed lines
    scope: global
    kind: saturating
    entries: 2048             # Same number of entries as RDP table
    width: 3                  # Small counter (0-7) to indicate how many times this signature was bypassed
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {is_bypassed_fill: {}} # If the fill was bypassed (way == NUM_WAY)
    track_bypassed_signature: {table: bypassed_tracker, index: pc_sig} # Record the signature
    stop: true # No further processing for a bypassed fill
  - decrement_bypassed_tracker_if_present: {table: bypassed_tracker, index: pc_sig, threshold: 0} # If the line was inserted, and its signature was tracked, decrement the counter
  - run: access
===== NEW_WORDS =====

- name: track_bypassed_signature
  place: step
  params: {table: table, index: signature}
  means: >
    Increments the entry in the specified table at the given index. If the entry
    is absent, it is initialized to 1. This records that a line with this signature was bypassed.
- name: rdp_gt_threshold_if_not_tracked
  place: condition
  params: {rdp_table: table, rdp_index: signature, rdp_threshold: {int: [0, 127]}, tracker_table: table, tracker_index: signature, tracker_threshold: {int: [0, 127]}}
  means: >
    True if (the entry in `rdp_table` at `rdp_index` is present and its value is greater
    than `rdp_threshold`) AND (the entry in `tracker_table` at `tracker_index` is absent
    or its value is less than `tracker_threshold`). Otherwise, returns False.
- name: rdp_exceeds_victim_if_not_tracked
  place: condition
  params: {rdp_table: table, rdp_index: signature, div: {int: [1, 128]}, tracker_table: table, tracker_index: signature, tracker_threshold: {int: [0, 127]}}
  means: >
    True if (the `table_exceeds_victim` condition for `rdp_table`, `rdp_index`, and `div` holds)
    AND (the entry in `tracker_table` at `tracker_index` is absent or its value is less
    than `tracker_threshold`). Otherwise, returns False.
- name: decrement_bypassed_tracker_if_present
  place: step
  params: {table: table, index: signature, threshold: {int: [0, 127]}}
  means: >
    If the entry in `table` at `index` is present and its value is greater than `threshold`,
    decrement it by one. This consumes a "wrong bypass" signal when the line is finally inserted.
