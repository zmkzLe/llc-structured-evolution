===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bh
doc: >
  Mockingjay with a Bypass History (BH) table. This design addresses the wrong_bypass
  problem by learning from past bypass mistakes. A new 2-bit table, bypass_history,
  tracks signatures of bypassed lines. It is incremented on a bypass and decremented
  on a hit. The bypass decision is now smarter: a line is only bypassed if the RDP
  predicts it's a scan AND the bypass_history table shows no evidence of prior
  wrong bypasses for that signature.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:
    - words:
        rdp_predicts_scan_and_no_bad_history:
          rdp_table: rdp
          history_table: bypass_history
          index: pc_sig
          rdp_threshold: 73
    - table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95
    unset: true
  - name: bypass_history
    scope: global
    kind: saturating
    entries: 2048
    width: 2
    init: ZERO

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - train: {table: bypass_history, index: pc_sig, op: decrement}
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {words: {is_bypassed_fill: {}}}
    train: {table: bypass_history, index: pc_sig, op: increment}
  - run: access
===== NEW_WORDS =====

- name: is_bypassed_fill
  place: condition
  params: {}
  means: >
    True when the current on_fill event has the way parameter equal to NUM_WAY, indicating that the
    fill was bypassed by the replacement policy.
- name: rdp_predicts_scan_and_no_bad_history
  place: condition
  params: {rdp_table: table, history_table: table, index: signature, rdp_threshold: {int: [0, 127]}}
  means: >
    True only if the value in the rdp_table at the given index is greater than the rdp_threshold,
    and the value in the history_table at the same index is zero. An absent entry in either
    table is treated as zero, so this condition would be false if the rdp_table entry is absent.
