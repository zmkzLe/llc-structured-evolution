===== STRUCTURE_DESCRIPTION =====
# Reference: DPC4-ChampSim/replacement/mockingjay, single-core build. Sizes are for
# the DPC4 LLC, 4096 x 12: INF_RD 95, INF_ETR 11, MAX_RD 73, 11-bit signatures,
# 32 sampled sets. The multi-core build folds the core id instead of the hit bit,
# inserts an unknown signature at INF_ETR, and scales prefetch samples by less.
name: mockingjay
doc: >
  Mockingjay: each line holds an estimated time to reuse (ETR), counted down every
  8 accesses to its set; the victim is the line with the largest |ETR|, an overdue
  (negative) one on ties. A predictor table keyed by the hit-or-miss PC signature
  gives the ETR on insertion, and a fill predicted to be reused after every line in
  the set is bypassed. A sampled shadow cache on 32 sets measures reuse distances
  with an 8-bit per-set clock and trains the predictor one step at a time; entries
  older than INF_RD train it up as scans. Writebacks are inserted at -INF_ETR and
  train nothing.

  This design adds a small `bypassed_history` sampler to track bypassed lines and
  train the RDP. This addresses `wrong_bypass` errors by decrementing the RDP if a
  bypassed line is later reused, and incrementing if it's evicted from the tracker
  without reuse. A global clock was added for the tracker's LRU policy.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO
  - name: global_clock        # NEW: Global clock for bypassed_history sampler's age
    scope: global
    kind: counter
    width: 64
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

  - name: bypassed_history     # NEW: Sampler to track bypassed lines and provide feedback to RDP
    scope: global
    sets: {mirror: 32}        # Sampled sets for bypassed history, same as sampled_cache
    extra_index_bits: 0       # No extra indexing for simplicity
    ways: 1                   # Smallest possible: 1 way per sampler set
    tag_bits: 46              # Full line address tag for precise matching
    fields:
      - {name: pc_sig_bypassed, kind: counter, width: 11, init: ZERO}
      - {name: timestamp_bypassed, kind: counter, width: 64, init: ZERO}
    age: {clock: global_clock, stamp: timestamp_bypassed}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true} # LRU victim policy
    on_match:                 # A previously bypassed line was subsequently reused (wrong_bypass)
      - train: {table: rdp, index: pc_sig_bypassed, op: decrement} # Penalize bypass decision
      - free: true            # Remove from tracker as it's no longer 'bypassed and un-reused'
    on_replace:               # A bypassed line was evicted from tracker without being reused (good_bypass)
      - train: {table: rdp, index: pc_sig_bypassed, op: increment} # Reward bypass decision
    install: {pc_sig_bypassed: {signature: pc_sig}, timestamp_bypassed: {tick: global_clock}}


procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_access:
  - set: {global_clock: {add: 1}} # NEW: Increment global clock on every access
  - sample: bypassed_history  # NEW: Check if current access hits a bypassed entry
  - run: access               # Original Mockingjay access procedure

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {type_not_in: [WRITE]} # NEW: Only non-writes can be bypassed by policy
    when: {way: NUM_WAY}      # NEW: If the fill was bypassed
    sample: bypassed_history  # NEW: Install the bypassed line into the tracker
  - run: access
