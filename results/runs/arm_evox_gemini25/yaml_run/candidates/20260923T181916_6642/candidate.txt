===== STRUCTURE_DESCRIPTION =====
name: my_bypass_fix_v1
doc: >
  Mockingjay with a new mechanism to track and learn from wrong_bypass errors by using a Bypass Revisit Table (BRT).
  A new signature, `pc_sig_no_hit`, is used for BRT lookups to track lines regardless of their hit status.
  The BRT records `pc_sig_no_hit` for bypassed lines. Subsequently, if these lines are later filled or hit in the cache,
  the BRT triggers a decrement of the main RDP table (using the original `pc_sig` for context), thus refining the bypass prediction.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11
  - name: pc_sig_no_hit # New signature for bypass tracking, excludes 'hit' bit to match across hit/miss
    source: ip
    fold: [prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_revisit_table # New table to track recently bypassed lines
    scope: global
    kind: saturating
    entries: 256              # A small table to fit within metadata budget
    width: 3                  # Counter for how many times we've seen this bypassed/revisited
    init: MID                 # Initialize to a neutral value (3 for width 3)
    unset: true               # Entries can be unset
    max: 7                    # Max value for a 3-bit counter

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - check_bht: {bht_signature: pc_sig_no_hit, rdp_signature: pc_sig, event_type: HIT_IN_CACHE} # Check BHT on every non-write hit
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  - when: {way: NUM_WAY} # This means the fill was bypassed
    check_bht: {bht_signature: pc_sig_no_hit, rdp_signature: pc_sig, event_type: BYPASS_FILL}
    stop: true # Stop here for bypassed fills, do not run 'access'
  - check_bht: {bht_signature: pc_sig_no_hit, rdp_signature: pc_sig, event_type: FILL_IN_CACHE} # Not bypassed, so it's a regular fill
  - run: access # Original Mockingjay logic for regular fills.
===== NEW_WORDS =====

- name: check_bht
  place: step
  params:
    bht_signature: signature       # Signature for lookup in bypass_revisit_table (e.g., pc_sig_no_hit)
    rdp_signature: signature       # Signature for update in rdp table (e.g., pc_sig)
    event_type: {symbol: BHTEventType}
  means: >
    This step operates on the 'bypass_revisit_table' and the main 'rdp' table.
    It looks up 'bht_signature' in the 'bypass_revisit_table'.
    If 'event_type' is 'BYPASS_FILL': The 'bypass_revisit_table' entry for 'bht_signature' is incremented, clamped at its maximum. If the entry was unset, it is set to 1.
    If 'event_type' is 'HIT_IN_CACHE': If the 'bypass_revisit_table' entry for 'bht_signature' exists and is not ZERO, it implies a wrong_bypass. The 'rdp' table entry for 'rdp_signature' is decremented. The 'bypass_revisit_table' entry for 'bht_signature' is then reset to ZERO.
    If 'event_type' is 'FILL_IN_CACHE': If the 'bypass_revisit_table' entry for 'bht_signature' exists and its value is greater than its initial value (MID), it implies a wrong_bypass (too many bypasses for a line that eventually got filled). The 'rdp' table entry for 'rdp_signature' is decremented. The 'bypass_revisit_table' entry for 'bht_signature' is then reset to ZERO.
- name: BHTEventType
  place: symbol_list
  members:
    - BYPASS_FILL
    - HIT_IN_CACHE
    - FILL_IN_CACHE
