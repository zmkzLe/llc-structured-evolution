===== STRUCTURE_DESCRIPTION =====
name: mockingjay_bypass_learning
doc: >
  Mockingjay with bypass learning: adds a bypass_confidence table to track the
  effectiveness of bypass decisions per PC signature. This table is incremented
  on bypassed fills and decremented on hits/non-bypassed fills for highly confident
  bypassed PCs. The bypass decision now also considers this confidence.

state:
  - name: etr
    scope: per_line
    kind: signed
    width: 5                  # holds -11..11
    init: 0
  - name: etr_clock
    scope: per_set
    kind: counter
    width: 4
    init: 8                   # GRANULARITY: the first access to a set ages it
  - name: set_clock
    scope: per_set
    kind: counter
    width: 8                  # TIMESTAMP_BITS; wraps by design
    init: ZERO

victim:
  key: etr
  extremum: max_magnitude
  tie_break: prefer_negative
  on_search: none
  bypass_when:                # the incoming line is predicted to be reused after every resident
    - {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
    - {table_exceeds_victim: {table: rdp, index: pc_sig, div: 8}}
    # New bypass condition: high bypass_confidence for this PC_sig
    - {table: {table: bypass_confidence, index: pc_sig, cmp: gte, threshold: 4}} # threshold 4 (MID for 3-bit counter)

signatures:
  - name: pc_sig
    source: ip
    fold: [hit, prefetch]
    hash: crc3
    low_bits: 11

tables:
  - name: rdp
    scope: global
    kind: saturating
    entries: 2048
    width: 7
    max: 95                   # INF_RD
    unset: true
  - name: bypass_confidence # New table to track bypass effectiveness
    scope: global
    kind: saturating
    entries: 2048             # Same size as RDP for PC signatures
    width: 3                  # 0-7, 4 is MID
    init: MID                 # Start at a neutral confidence
    unset: true               # Allows tracking when an entry is first seen

samplers:
  - name: sampled_cache
    scope: global
    sets: {mirror: 32}
    extra_index_bits: 4       # 16 sampler sets per sampled set
    ways: 5
    tag_bits: 10
    fields:
      - {name: signature, kind: counter, width: 11, init: ZERO}
      - {name: timestamp, kind: counter, width: 8, init: ZERO}
    age: {clock: set_clock, stamp: timestamp}
    victim: {key: age, extremum: max, tie_break: first, prefer_free: true}
    on_match:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        stop: true            # too old to be a reuse distance; the sweep detrains it
      - when: {type_in: [PREFETCH]}
        train: {table: rdp, index: signature, op: toward, sample: age, scale: 2, min_diff: 16}
        free: true
        stop: true
      - train: {table: rdp, index: signature, op: toward, sample: age, min_diff: 16}
        free: true
    sweep:
      - when: {counter: {counter: age, cmp: gt, threshold: 95}}
        train: {table: rdp, index: signature, op: increment}
        free: true
    on_replace:
      - train: {table: rdp, index: signature, op: increment}
    install: {signature: {signature: pc_sig}, timestamp: {read: set_clock}}
    after:
      - set: {set_clock: {add: 1}}

procedures:
  access:                     # mockingjay.cc access(): the update body shared by hits and fills
    - sample: sampled_cache
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      each_way:
        except_touched: true
        when: {counter: {counter: etr, cmp: lt, threshold: 11, abs: true}}
        set: {etr: {add: -1}}
    - when: {counter: {counter: etr_clock, cmp: eq, threshold: 8}}
      set: {etr_clock: {const: 0}}
    - set: {etr_clock: {add: 1}}
    # New training for bypass_confidence: decrement on hit or non-bypassed fill for confident PCs
    - when: {table: {table: bypass_confidence, index: pc_sig, cmp: gte, threshold: 4}} # If this PC was considered a good bypass candidate
      train: {table: bypass_confidence, index: pc_sig, op: decrement} # Decrement, as it was actually useful
    - when: {table: {table: rdp, index: pc_sig, cmp: absent}}
      set: {etr: {const: 0}}
      stop: true
    - when: {table: {table: rdp, index: pc_sig, cmp: gt, threshold: 73}}
      set: {etr: {const: 11}}
      stop: true
    - set: {etr: {table: rdp, index: pc_sig, div: 8}}

on_hit:
  - when: {type_in: [WRITE]}
    stop: true
  - run: access

on_fill:
  - when: {type_in: [WRITE]}
    set: {etr: {const: -11}}
    stop: true
  # New training for bypass_confidence: increment on bypassed fill
  - when: {way: NUM_WAY} # This condition checks if the fill was bypassed
    train: {table: bypass_confidence, index: pc_sig, op: increment} # Increment confidence for this PC_sig
    stop: true # Skip normal fill logic if it was bypassed and we just trained
  - run: access
